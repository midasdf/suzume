# zz Phase 2: Tree-sitter Syntax Highlighting — Implementation Plan

> **For agentic workers:** REQUIRED: Use superpowers:subagent-driven-development (if subagents available) or superpowers:executing-plans to implement this plan. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Add syntax highlighting to zz using Tree-sitter incremental parsing, with Catppuccin Mocha syntax colors.

**Architecture:** Tree-sitter C library linked via system packages. Language grammars loaded at runtime via dlopen (system .so files at `/usr/lib/libtree-sitter-*.so`). Highlight queries (.scm) read from `/usr/share/tree-sitter/queries/`. A new `highlight.zig` module wraps the Tree-sitter C API and provides per-byte color lookup. The existing `view.zig` render pipeline queries highlight colors instead of using a single foreground color.

**Tech Stack:** libtree-sitter (system), libtree-sitter-{c,python,rust,javascript,bash} (system .so), Zig @cImport, highlight query files (.scm)

**Spec:** `docs/superpowers/specs/2026-03-31-zz-editor-design.md` (Section 3)

### Task 2: highlight.zig — Tree-sitter wrapper

**Files:**
- Create: `zz/src/editor/highlight.zig`

This is the core module. It needs to:

1. **@cImport** tree-sitter API + grammar headers
2. **Language detection** from file extension
3. **Parser management** — create parser, set language, parse with PieceTable read callback
4. **Query management** — load highlights.scm, create TSQuery
5. **Highlight lookup** — given a byte offset, return the syntax capture name (or null)
6. **Incremental re-parse** — on buffer edit, call ts_tree_edit + re-parse

- [ ] **Step 1: Write highlight.zig with C imports and types**

```zig
    @cInclude("tree_sitter/tree-sitter-c.h");
    @cInclude("tree_sitter/tree-sitter-python.h");
    @cInclude("tree_sitter/tree-sitter-rust.h");
    @cInclude("tree_sitter/tree-sitter-javascript.h");
    @cInclude("tree_sitter/tree-sitter-bash.h");
```

Define a `SyntaxKind` enum mapping Tree-sitter capture names to categories:
```zig
pub const SyntaxKind = enum {
    keyword,
    function,
    function_builtin,
    type_name,
    string,
    number,
    comment,
    operator,
    variable,
    constant,
    property,
    punctuation,
    none,

    pub fn fromCaptureName(name: []const u8) SyntaxKind {
        if (startsWith(name, "keyword")) return .keyword;
        if (startsWith(name, "function.builtin")) return .function_builtin;
        if (startsWith(name, "function")) return .function;
        if (startsWith(name, "type")) return .type_name;
        if (startsWith(name, "string")) return .string;
        if (startsWith(name, "number") or startsWith(name, "float")) return .number;
        if (startsWith(name, "comment")) return .comment;
        if (startsWith(name, "operator")) return .operator;
        if (startsWith(name, "variable")) return .variable;
        if (startsWith(name, "constant")) return .constant;
        if (startsWith(name, "property") or startsWith(name, "field")) return .property;
        if (startsWith(name, "punctuation")) return .punctuation;
        return .none;
    }

    fn startsWith(haystack: []const u8, prefix: []const u8) bool {
        return haystack.len >= prefix.len and std.mem.eql(u8, haystack[0..prefix.len], prefix);
    }
};
```

- [ ] **Step 3: Implement setLanguage — detect from file extension, load grammar + query**

```zig
    pub fn setLanguage(self: *Highlighter, file_path: ?[]const u8) void {
        const ext = getExtension(file_path orelse return) orelse return;

        const lang: ?*const ts.TSLanguage = if (std.mem.eql(u8, ext, "c") or std.mem.eql(u8, ext, "h"))
            ts.tree_sitter_c()
        else if (std.mem.eql(u8, ext, "py"))
            ts.tree_sitter_python()
        else if (std.mem.eql(u8, ext, "rs"))
            ts.tree_sitter_rust()
        else if (std.mem.eql(u8, ext, "js") or std.mem.eql(u8, ext, "jsx"))
            ts.tree_sitter_javascript()
        else if (std.mem.eql(u8, ext, "sh") or std.mem.eql(u8, ext, "bash"))
            ts.tree_sitter_bash()
        else
            null;

        self.language = lang;
        if (lang) |l| {
            _ = ts.ts_parser_set_language(self.parser, l);
            self.loadQuery(ext);
        }
```

For the query loading, read the .scm file from system paths:
```zig
    fn loadQuery(self: *Highlighter, ext: []const u8) void {
        if (self.query) |q| { ts.ts_query_delete(q); self.query = null; }

        const lang_name = extToLangName(ext) orelse return;
        var path_buf: [256]u8 = undefined;
        const query_path = std.fmt.bufPrint(&path_buf, "/usr/share/tree-sitter/queries/{s}/highlights.scm", .{lang_name}) catch return;

        const query_src = std.fs.cwd().readFileAlloc(self.allocator, query_path, 1024 * 1024) catch return;
        defer self.allocator.free(query_src);

        var error_offset: u32 = 0;
        var error_type: ts.TSQueryError = ts.TSQueryErrorNone;
        self.query = ts.ts_query_new(
            self.language.?,
            query_src.ptr,
            @intCast(query_src.len),
            &error_offset,
            &error_type,
        );
    }

    fn extToLangName(ext: []const u8) ?[]const u8 {
        if (std.mem.eql(u8, ext, "c") or std.mem.eql(u8, ext, "h")) return "c";
        if (std.mem.eql(u8, ext, "py")) return "python";
        if (std.mem.eql(u8, ext, "rs")) return "rust";
        if (std.mem.eql(u8, ext, "js") or std.mem.eql(u8, ext, "jsx")) return "javascript";
        if (std.mem.eql(u8, ext, "sh") or std.mem.eql(u8, ext, "bash")) return "bash";
        return null;
    }

    fn getExtension(path: []const u8) ?[]const u8 {
        const idx = std.mem.lastIndexOfScalar(u8, path, '.') orelse return null;
        return path[idx + 1 ..];
    }
```

- [ ] **Step 4: Implement parse — PieceTable read callback + full parse**

The key bridge between Piece Table and Tree-sitter:

```zig
    pub fn parse(self: *Highlighter, buffer: *const PieceTable) void {
        if (self.language == null) return;

        const input = ts.TSInput{
            .payload = @constCast(@ptrCast(buffer)),
            .read = tsReadCallback,
            .encoding = ts.TSInputEncodingUTF8,
        };

        if (self.tree) |old_tree| {
            self.tree = ts.ts_parser_parse(self.parser, old_tree, input);
            ts.ts_tree_delete(old_tree);
        } else {
            self.tree = ts.ts_parser_parse(self.parser, null, input);
        }
    }

    fn tsReadCallback(payload: ?*anyopaque, byte_index: u32, _: ts.TSPoint, bytes_read: *u32) callconv(.C) [*c]const u8 {
        const buffer: *const PieceTable = @ptrCast(@alignCast(payload));
        const slice = buffer.contiguousSliceAt(byte_index);
        bytes_read.* = @intCast(slice.len);
        if (slice.len == 0) return null;
        return slice.ptr;
    }
```

- [ ] **Step 5: Implement notifyEdit — for incremental re-parsing**

```zig
    /// Call after buffer edit to keep tree in sync.
    /// start_byte: byte offset where edit starts
    /// old_end_byte: byte offset where old content ended
    /// new_end_byte: byte offset where new content ends
    pub fn notifyEdit(self: *Highlighter, buffer: *const PieceTable, start_byte: u32, old_end_byte: u32, new_end_byte: u32) void {
        if (self.tree) |tree| {
            const start_pos = buffer.offsetToLineCol(start_byte);
            const new_end_pos = buffer.offsetToLineCol(new_end_byte);

            const edit = ts.TSInputEdit{
                .start_byte = start_byte,
                .old_end_byte = old_end_byte,
                .new_end_byte = new_end_byte,
                .start_point = .{ .row = start_pos.line, .column = start_pos.col },
                .old_end_point = .{ .row = 0, .column = 0 },  // Approximation OK for re-parse
                .new_end_point = .{ .row = new_end_pos.line, .column = new_end_pos.col },
            };
            ts.ts_tree_edit(tree, &edit);
        }
        self.parse(buffer);
    }
```

- [ ] **Step 6: Implement getSyntaxAt — query highlight for a byte range**

This is the main API used by view.zig during rendering. For a given visible line range, execute the highlight query and return syntax kinds.

Rather than querying per-byte (too slow), query per visible range and cache:

```zig
    /// Get syntax kind for a byte offset. Uses last query result cache.
    /// Call queryRange() before accessing.
    pub fn getSyntaxAt(self: *Highlighter, byte_offset: u32) SyntaxKind {
        if (self.tree == null or self.query == null) return .none;

        // Binary search in cached highlights
        for (self.cached_highlights.items) |h| {
            if (byte_offset >= h.start and byte_offset < h.end) return h.kind;
        }
        return .none;
    }

    /// Query highlights for a byte range (call before rendering visible lines).
    pub fn queryRange(self: *Highlighter, start_byte: u32, end_byte: u32) void {
        self.cached_highlights.clearRetainingCapacity();

        const tree = self.tree orelse return;
        const query = self.query orelse return;

        const cursor = ts.ts_query_cursor_new() orelse return;
        defer ts.ts_query_cursor_delete(cursor);

        ts.ts_query_cursor_set_byte_range(cursor, start_byte, end_byte);
        ts.ts_query_cursor_exec(cursor, query, ts.ts_tree_root_node(tree));

        var match: ts.TSQueryMatch = undefined;
        var capture_index: u32 = 0;
        while (ts.ts_query_cursor_next_capture(cursor, &match, &capture_index)) {
            const capture = match.captures[capture_index];
            const node = capture.node;
            const start = ts.ts_node_start_byte(node);
            const end = ts.ts_node_end_byte(node);

            var name_len: u32 = 0;
            const name_ptr = ts.ts_query_capture_name_for_id(query, capture.index, &name_len);
            if (name_ptr == null) continue;
            const name = name_ptr[0..name_len];
            const kind = SyntaxKind.fromCaptureName(name);
            if (kind == .none) continue;

            self.cached_highlights.append(self.allocator, .{
                .start = start,
                .end = end,
                .kind = kind,
            }) catch {};
        }
    }
```

Add to struct fields:
```zig
    cached_highlights: std.ArrayList(Highlight),

    const Highlight = struct {
        start: u32,
        end: u32,
        kind: SyntaxKind,
    };
```

And update init/deinit to manage cached_highlights.
