# zz Phase 1: Minimal Editor — Implementation Plan

> **For agentic workers:** REQUIRED: Use superpowers:subagent-driven-development (if subagents available) or superpowers:executing-plans to implement this plan. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Build a single-file code editor with xcb+SHM rendering, freetype fonts, Piece Table buffer, undo/redo, clipboard, XIM, and file I/O.

**Architecture:** xcb window with SHM double-buffering (zt pattern). Freetype for glyph rendering with ASCII pre-cache. Piece Table for O(1) insert text buffer. Single-threaded epoll event loop. All rendering event-driven — no idle CPU usage.

**Tech Stack:** Zig 0.15+, xcb/xcb-shm/xcb-xkb/xkbcommon/xcb-imdkit (from zt), freetype2 (from suzume), no Zig package dependencies.

**Spec:** `docs/superpowers/specs/2026-03-31-zz-editor-design.md`

// zz/src/editor/buffer.zig
const std = @import("std");

pub const PieceTable = struct {
    original: []const u8,           // Immutable source (mmap'd or owned)
    add_buf: std.ArrayList(u8),     // Append-only insert buffer
    pieces: std.ArrayList(Piece),   // Ordered piece sequence
    total_len: u32,                 // Cached total document length
    allocator: std.mem.Allocator,

    pub const Source = enum { original, add };

    pub const Piece = struct {
        source: Source,
        start: u32,
        len: u32,
        newline_count: u32,
    };

    pub fn init(allocator: std.mem.Allocator, content: []const u8) !PieceTable {
        var pieces = std.ArrayList(Piece).init(allocator);
        if (content.len > 0) {
            try pieces.append(.{
                .source = .original,
                .start = 0,
                .len = @intCast(content.len),
                .newline_count = countNewlines(content),
            });
        }
        return .{
            .original = content,
            .add_buf = std.ArrayList(u8).init(allocator),
            .pieces = pieces,
            .total_len = @intCast(content.len),
            .allocator = allocator,
        };
    }

    pub fn deinit(self: *PieceTable) void {
        self.pieces.deinit();
        self.add_buf.deinit();
    }

    pub fn countNewlines(data: []const u8) u32 {
        var count: u32 = 0;
        for (data) |byte| {
            if (byte == '\n') count += 1;
        }
        return count;
    }

    /// Get the content of a piece as a slice.
    pub fn pieceContent(self: *const PieceTable, piece: Piece) []const u8 {
        const buf = switch (piece.source) {
            .original => self.original,
            .add => self.add_buf.items,
        };
        return buf[piece.start..][0..piece.len];
    }

    /// Total number of lines (newline_count + 1).
    pub fn lineCount(self: *const PieceTable) u32 {
        var total: u32 = 0;
        for (self.pieces.items) |p| {
            total += p.newline_count;
        }
        return total + 1;
    }
};
```

/// Insert text at byte offset `pos`.
/// Returns the index of the first new/modified piece for dirty tracking.
pub fn insert(self: *PieceTable, pos: u32, text: []const u8) !void {
    if (text.len == 0) return;
    if (pos > self.total_len) return error.OutOfBounds;

    const add_start: u32 = @intCast(self.add_buf.items.len);
    try self.add_buf.appendSlice(text);

    const new_piece = Piece{
        .source = .add,
        .start = add_start,
        .len = @intCast(text.len),
        .newline_count = countNewlines(text),
    };

    if (self.pieces.items.len == 0) {
        try self.pieces.append(new_piece);
        self.total_len += @intCast(text.len);
        return;
    }

    // Find which piece contains `pos`
    var offset: u32 = 0;
    var idx: usize = 0;
    for (self.pieces.items, 0..) |p, i| {
        if (offset + p.len > pos or i == self.pieces.items.len - 1) {
            idx = i;
            break;
        }
        offset += p.len;
    }

    const piece = self.pieces.items[idx];
    const local_pos = pos - offset;

    if (local_pos == 0) {
        // Insert before this piece
        try self.pieces.insert(idx, new_piece);
    } else if (local_pos == piece.len) {
        // Insert after this piece
        try self.pieces.insert(idx + 1, new_piece);
    } else {
        // Split piece: [before | new_piece | after]
        const before = Piece{
            .source = piece.source,
            .start = piece.start,
            .len = local_pos,
            .newline_count = countNewlines(self.pieceContent(piece)[0..local_pos]),
        };
        const after = Piece{
            .source = piece.source,
            .start = piece.start + local_pos,
            .len = piece.len - local_pos,
            .newline_count = countNewlines(self.pieceContent(piece)[local_pos..]),
        };
        self.pieces.items[idx] = before;
        try self.pieces.insert(idx + 1, new_piece);
        try self.pieces.insert(idx + 2, after);
    }

    self.total_len += @intCast(text.len);
}
```

/// Write entire document content to a writer.
pub fn writeAll(self: *const PieceTable, writer: anytype) !void {
    for (self.pieces.items) |piece| {
        try writer.writeAll(self.pieceContent(piece));
    }
}

/// Collect entire document content into an allocated slice.
pub fn collectContent(self: *const PieceTable, allocator: std.mem.Allocator) ![]u8 {
    var result = try allocator.alloc(u8, self.total_len);
    var offset: usize = 0;
    for (self.pieces.items) |piece| {
        const content = self.pieceContent(piece);
        @memcpy(result[offset..][0..content.len], content);
        offset += content.len;
    }
    return result;
}

/// Delete `len` bytes starting at byte offset `pos`.
pub fn delete(self: *PieceTable, pos: u32, len: u32) !void {
    if (len == 0) return;
    if (pos + len > self.total_len) return error.OutOfBounds;

    const end = pos + len;
    var new_pieces = std.ArrayList(Piece).init(self.allocator);
    errdefer new_pieces.deinit();

    var offset: u32 = 0;
    for (self.pieces.items) |piece| {
        const piece_start = offset;
        const piece_end = offset + piece.len;
        offset = piece_end;

        if (piece_end <= pos or piece_start >= end) {
            // Entirely outside deletion range — keep as-is
            try new_pieces.append(piece);
        } else {
            // Overlaps with deletion range
            if (piece_start < pos) {
                // Keep left portion
                const keep_len = pos - piece_start;
                try new_pieces.append(.{
                    .source = piece.source,
                    .start = piece.start,
                    .len = keep_len,
                    .newline_count = countNewlines(self.pieceContent(piece)[0..keep_len]),
                });
            }
            if (piece_end > end) {
                // Keep right portion
                const skip = end - piece_start;
                const keep_len = piece_end - end;
                try new_pieces.append(.{
                    .source = piece.source,
                    .start = piece.start + skip,
                    .len = keep_len,
                    .newline_count = countNewlines(self.pieceContent(piece)[skip..]),
                });
            }
        }
    }

    self.pieces.deinit();
    self.pieces = new_pieces;
    self.total_len -= len;
}
```

/// Return a contiguous slice starting at byte offset within a single piece.
/// Used by Tree-sitter read callback — returns only up to piece boundary.
pub fn contiguousSliceAt(self: *const PieceTable, byte_offset: u32) []const u8 {
    var offset: u32 = 0;
    for (self.pieces.items) |piece| {
        if (offset + piece.len > byte_offset) {
            const local = byte_offset - offset;
            const content = self.pieceContent(piece);
            return content[local..];
        }
        offset += piece.len;
    }
    return "";
}

/// Convert a line number (0-based) to byte offset of line start.
pub fn lineToOffset(self: *const PieceTable, target_line: u32) u32 {
    if (target_line == 0) return 0;

    var line: u32 = 0;
    var offset: u32 = 0;
    for (self.pieces.items) |piece| {
        if (line + piece.newline_count >= target_line) {
            // Target line starts within this piece
            const content = self.pieceContent(piece);
            for (content) |byte| {
                if (byte == '\n') {
                    line += 1;
                    if (line == target_line) return offset + 1;
                }
                offset += 1;
            }
        } else {
            line += piece.newline_count;
            offset += piece.len;
        }
    }
    return offset;
}

/// Convert a byte offset to (line, col) both 0-based.
pub fn offsetToLineCol(self: *const PieceTable, target_offset: u32) struct { line: u32, col: u32 } {
    var line: u32 = 0;
    var col: u32 = 0;
    var offset: u32 = 0;
    for (self.pieces.items) |piece| {
        const content = self.pieceContent(piece);
        for (content) |byte| {
            if (offset == target_offset) return .{ .line = line, .col = col };
            if (byte == '\n') {
                line += 1;
                col = 0;
            } else {
                col += 1;
            }
            offset += 1;
        }
    }
    return .{ .line = line, .col = col };
}
```

```zig
pub fn undo(self: *PieceTable) !bool {
    if (self.undo_stack.items.len == 0) return false;

    // Save current state to redo
    const current_snapshot = try self.allocator.dupe(Piece, self.pieces.items);
    try self.redo_stack.append(.{
        .pieces_snapshot = current_snapshot,
        .total_len = self.total_len,
    });

    // Restore from undo stack
    var entry = self.undo_stack.pop();
    self.pieces.clearRetainingCapacity();
    try self.pieces.appendSlice(entry.pieces_snapshot);
    self.total_len = entry.total_len;
    entry.deinit(self.allocator);

    return true;
}

pub fn redo(self: *PieceTable) !bool {
    if (self.redo_stack.items.len == 0) return false;

    // Save current state to undo
    const current_snapshot = try self.allocator.dupe(Piece, self.pieces.items);
    try self.undo_stack.append(.{
        .pieces_snapshot = current_snapshot,
        .total_len = self.total_len,
    });

    // Restore from redo stack
    var entry = self.redo_stack.pop();
    self.pieces.clearRetainingCapacity();
    try self.pieces.appendSlice(entry.pieces_snapshot);
    self.total_len = entry.total_len;
    entry.deinit(self.allocator);

    return true;
}

pub const Selection = struct {
    /// Anchor position (start of selection, stays fixed).
    anchor: u32,
    /// Head position (moves with cursor).
    head: u32,

    pub fn hasSelection(self: Selection) bool {
        return self.anchor != self.head;
    }

    pub fn start(self: Selection) u32 {
        return @min(self.anchor, self.head);
    }

    pub fn end(self: Selection) u32 {
        return @max(self.anchor, self.head);
    }

    pub fn collapse(self: *Selection) void {
        self.anchor = self.head;
    }
};

pub const CursorState = struct {
    cursors: std.ArrayList(Selection),
    desired_col: ?u32 = null,  // Sticky column for vertical movement

    pub fn init(allocator: std.mem.Allocator) CursorState {
        var cursors = std.ArrayList(Selection).init(allocator);
        // Always have at least one cursor
        cursors.append(.{ .anchor = 0, .head = 0 }) catch unreachable;
        return .{ .cursors = cursors, .desired_col = null };
    }

    pub fn deinit(self: *CursorState) void {
        self.cursors.deinit();
    }

    /// Get the primary (first) cursor.
    pub fn primary(self: *const CursorState) Selection {
        return self.cursors.items[0];
    }

    /// Set primary cursor position, collapsing selection.
    pub fn moveTo(self: *CursorState, pos: u32) void {
        self.cursors.items[0] = .{ .anchor = pos, .head = pos };
        self.desired_col = null;  // Reset sticky column
    }

    /// Extend selection from anchor to new head.
    pub fn selectTo(self: *CursorState, pos: u32) void {
        self.cursors.items[0].head = pos;
    }

    /// Move cursor right by one character, respecting document bounds.
    pub fn moveRight(self: *CursorState, buf: *const PieceTable, extend: bool) void {
        const cur = &self.cursors.items[0];
        if (cur.head < buf.total_len) {
            // Skip UTF-8 continuation bytes
            const slice = buf.contiguousSliceAt(cur.head);
            const byte_len = utf8ByteLen(slice[0]);
            cur.head = @min(cur.head + byte_len, buf.total_len);
        }
        if (!extend) cur.anchor = cur.head;
        self.desired_col = null;
    }

    /// Move cursor left by one character.
    pub fn moveLeft(self: *CursorState, buf: *const PieceTable, extend: bool) void {
        const cur = &self.cursors.items[0];
        if (cur.head > 0) {
            // Walk back over UTF-8 continuation bytes
            var pos = cur.head - 1;
            while (pos > 0) {
                const slice = buf.contiguousSliceAt(pos);
                if (slice.len > 0 and (slice[0] & 0xC0) != 0x80) break;
                pos -= 1;
            }
            cur.head = pos;
        }
        if (!extend) cur.anchor = cur.head;
        self.desired_col = null;
    }

    /// Move cursor to start of line.
    pub fn moveHome(self: *CursorState, buf: *const PieceTable, extend: bool) void {
        const cur = &self.cursors.items[0];
        const lc = buf.offsetToLineCol(cur.head);
        cur.head = buf.lineToOffset(lc.line);
        if (!extend) cur.anchor = cur.head;
    }

    /// Move cursor to end of line.
    pub fn moveEnd(self: *CursorState, buf: *const PieceTable, extend: bool) void {
        const cur = &self.cursors.items[0];
        const lc = buf.offsetToLineCol(cur.head);
        const next_line_offset = buf.lineToOffset(lc.line + 1);
        // End of line = before the \n (or at document end)
        if (next_line_offset > 0 and lc.line + 1 < buf.lineCount()) {
            cur.head = next_line_offset - 1;
        } else {
            cur.head = buf.total_len;
        }
        if (!extend) cur.anchor = cur.head;
    }

    fn utf8ByteLen(first_byte: u8) u32 {
        if (first_byte < 0x80) return 1;
        if (first_byte < 0xE0) return 2;
        if (first_byte < 0xF0) return 3;
        return 4;
    }
};

// zz/src/ui/font.zig
const std = @import("std");

const c = @cImport({
    @cInclude("ft2build.h");
    @cInclude("freetype/freetype.h");
});

pub const GlyphBitmap = struct {
    pixels: []const u8,   // 8-bit alpha values (owned copy)
    width: u16,
    height: u16,
    bearing_x: i16,
    bearing_y: i16,
    advance: u16,         // Horizontal advance in pixels
};

pub const FontFace = struct {
    ft_lib: c.FT_Library,
    face: c.FT_Face,
    cell_width: u32,      // Max advance (monospace = all same)
    cell_height: u32,     // ascent + descent + line_gap
    ascent: i32,
    descent: i32,
    cache: std.AutoHashMap(u32, GlyphBitmap),
    allocator: std.mem.Allocator,

    pub fn init(allocator: std.mem.Allocator, font_path: [*:0]const u8, size_px: u32) !FontFace {
        var ft_lib: c.FT_Library = undefined;
        if (c.FT_Init_FreeType(&ft_lib) != 0) return error.FreetypeInitFailed;

        var face: c.FT_Face = undefined;
        if (c.FT_New_Face(ft_lib, font_path, 0, &face) != 0) return error.FontLoadFailed;
        if (c.FT_Set_Pixel_Sizes(face, 0, size_px) != 0) return error.FontSizeFailed;

        const metrics = face.*.size.*.metrics;
        const ascent = @divTrunc(metrics.ascender, 64);
        const descent = @divTrunc(-metrics.descender, 64);  // descender is negative
        const cell_height: u32 = @intCast(ascent + descent + 1);

        // For monospace: load 'M' to get advance width
        if (c.FT_Load_Char(face, 'M', c.FT_LOAD_DEFAULT) != 0) return error.GlyphLoadFailed;
        const cell_width: u32 = @intCast(@divTrunc(face.*.glyph.*.advance.x, 64));

        var font = FontFace{
            .ft_lib = ft_lib,
            .face = face,
            .cell_width = cell_width,
            .cell_height = cell_height,
            .ascent = @intCast(ascent),
            .descent = @intCast(descent),
            .cache = std.AutoHashMap(u32, GlyphBitmap).init(allocator),
            .allocator = allocator,
        };

        // Pre-cache ASCII
        for (0x20..0x7F) |cp| {
            _ = try font.getGlyph(@intCast(cp));
        }

        return font;
    }

    pub fn deinit(self: *FontFace) void {
        var it = self.cache.valueIterator();
        while (it.next()) |glyph| {
            self.allocator.free(glyph.pixels);
        }
        self.cache.deinit();
        _ = c.FT_Done_Face(self.face);
        _ = c.FT_Done_FreeType(self.ft_lib);
    }

    pub fn getGlyph(self: *FontFace, codepoint: u32) !GlyphBitmap {
        if (self.cache.get(codepoint)) |cached| return cached;

        if (c.FT_Load_Char(self.face, codepoint, c.FT_LOAD_RENDER | c.FT_LOAD_TARGET_LIGHT) != 0) {
            // Return empty glyph for missing characters
            const empty = GlyphBitmap{
                .pixels = &.{},
                .width = 0,
                .height = 0,
                .bearing_x = 0,
                .bearing_y = 0,
                .advance = @intCast(self.cell_width),
            };
            try self.cache.put(codepoint, empty);
            return empty;
        }

        const glyph = self.face.*.glyph;
        const bitmap = glyph.*.bitmap;
        const w: u16 = @intCast(bitmap.width);
        const h: u16 = @intCast(bitmap.rows);

        // Copy bitmap data (freetype reuses internal buffer)
        const pixel_count = @as(usize, w) * @as(usize, h);
        const pixels = if (pixel_count > 0) blk: {
            const p = try self.allocator.alloc(u8, pixel_count);
            // Copy row by row (pitch may differ from width)
            const src: [*]const u8 = bitmap.buffer;
            for (0..h) |row| {
                const src_offset = row * @as(usize, @intCast(bitmap.pitch));
                const dst_offset = row * @as(usize, w);
                @memcpy(p[dst_offset..][0..w], src[src_offset..][0..w]);
            }
            break :blk p;
        } else &[_]u8{};

        const result = GlyphBitmap{
            .pixels = pixels,
            .width = w,
            .height = h,
            .bearing_x = @intCast(glyph.*.bitmap_left),
            .bearing_y = @intCast(glyph.*.bitmap_top),
            .advance = @intCast(@divTrunc(glyph.*.advance.x, 64)),
        };

        try self.cache.put(codepoint, result);
        return result;
    }
};

// zz/src/ui/render.zig
const std = @import("std");
const FontFace = @import("font.zig").FontFace;
const GlyphBitmap = @import("font.zig").GlyphBitmap;

pub const Color = struct {
    r: u8,
    g: u8,
    b: u8,

    pub fn fromHex(hex: u24) Color {
        return .{
            .r = @intCast((hex >> 16) & 0xFF),
            .g = @intCast((hex >> 8) & 0xFF),
            .b = @intCast(hex & 0xFF),
        };
    }
};

pub const Renderer = struct {
    buffer: []u8,       // SHM buffer (BGRA32)
    stride: u32,        // bytes per row
    width: u32,         // pixel width
    height: u32,        // pixel height

    pub fn fillRect(self: *Renderer, x: u32, y: u32, w: u32, h: u32, color: Color) void {
        const pixel: u32 = @as(u32, color.b) |
            (@as(u32, color.g) << 8) |
            (@as(u32, color.r) << 16) |
            (0xFF << 24);
        const pixel_bytes: [4]u8 = @bitCast(pixel);

        var row = y;
        while (row < y + h and row < self.height) : (row += 1) {
            var col = x;
            while (col < x + w and col < self.width) : (col += 1) {
                const offset = row * self.stride + col * 4;
                if (offset + 4 <= self.buffer.len) {
                    self.buffer[offset..][0..4].* = pixel_bytes;
                }
            }
        }
    }

    /// Render a glyph at pixel position (px, py) with alpha blending.
    pub fn drawGlyph(self: *Renderer, glyph: GlyphBitmap, px: i32, py: i32, fg: Color) void {
        if (glyph.width == 0 or glyph.height == 0) return;

        var row: u32 = 0;
        while (row < glyph.height) : (row += 1) {
            const screen_y = py + @as(i32, @intCast(row));
            if (screen_y < 0 or screen_y >= @as(i32, @intCast(self.height))) continue;

            var col: u32 = 0;
            while (col < glyph.width) : (col += 1) {
                const screen_x = px + @as(i32, @intCast(col));
                if (screen_x < 0 or screen_x >= @as(i32, @intCast(self.width))) continue;

                const alpha = glyph.pixels[row * glyph.width + col];
                if (alpha == 0) continue;

                const offset = @as(u32, @intCast(screen_y)) * self.stride + @as(u32, @intCast(screen_x)) * 4;
                if (offset + 4 > self.buffer.len) continue;

                if (alpha == 255) {
                    self.buffer[offset] = fg.b;
                    self.buffer[offset + 1] = fg.g;
                    self.buffer[offset + 2] = fg.r;
                    self.buffer[offset + 3] = 0xFF;
                } else {
                    // Alpha blend
                    const a = @as(u16, alpha);
                    const inv_a = 255 - a;
                    self.buffer[offset] = @intCast((@as(u16, fg.b) * a + @as(u16, self.buffer[offset]) * inv_a) / 255);
                    self.buffer[offset + 1] = @intCast((@as(u16, fg.g) * a + @as(u16, self.buffer[offset + 1]) * inv_a) / 255);
                    self.buffer[offset + 2] = @intCast((@as(u16, fg.r) * a + @as(u16, self.buffer[offset + 2]) * inv_a) / 255);
                }
            }
        }
    }

    /// Render a single character in a grid cell.
    pub fn drawCell(
        self: *Renderer,
        font: *FontFace,
        codepoint: u32,
        grid_col: u32,
        grid_row: u32,
        fg: Color,
        bg: Color,
        left_pad: u32,
    ) void {
        const cell_x = left_pad + grid_col * font.cell_width;
        const cell_y = grid_row * font.cell_height;

        // Fill background
        self.fillRect(cell_x, cell_y, font.cell_width, font.cell_height, bg);

        // Draw glyph
        const glyph = font.getGlyph(codepoint) catch return;
        const glyph_x = @as(i32, @intCast(cell_x)) + glyph.bearing_x;
        const glyph_y = @as(i32, @intCast(cell_y)) + font.ascent - glyph.bearing_y;
        self.drawGlyph(glyph, glyph_x, glyph_y, fg);
    }
};

const std = @import("std");
const PieceTable = @import("buffer.zig").PieceTable;
const CursorState = @import("cursor.zig").CursorState;
const Renderer = @import("../ui/render.zig").Renderer;
const FontFace = @import("../ui/font.zig").FontFace;
const Color = @import("../ui/render.zig").Color;

pub const EditorView = struct {
    buffer: PieceTable,
    cursor: CursorState,
    scroll_line: u32,           // First visible line (0-based)
    visible_rows: u32,          // Number of visible text rows
    visible_cols: u32,          // Number of visible columns
    left_pad: u32,              // Left padding in pixels
    dirty_rows: []bool,         // Per-row dirty flags
    allocator: std.mem.Allocator,
    modified: bool,             // Unsaved changes
    file_path: ?[]const u8,     // Opened file path (owned)

    // Theme colors (hardcoded for Phase 1, configurable later)
    const bg_color = Color.fromHex(0x1e1e2e);
    const fg_color = Color.fromHex(0xcdd6f4);
    const cursor_color = Color.fromHex(0xf5e0dc);
    const selection_color = Color.fromHex(0x45475a);
    const line_num_color = Color.fromHex(0x6c7086);
    const line_num_active_color = Color.fromHex(0xcdd6f4);

    pub fn init(allocator: std.mem.Allocator, content: []const u8) !EditorView {
        return .{
            .buffer = try PieceTable.init(allocator, content),
            .cursor = CursorState.init(allocator),
            .scroll_line = 0,
            .visible_rows = 0,
            .visible_cols = 0,
            .left_pad = 8,
            .dirty_rows = &.{},
            .allocator = allocator,
            .modified = false,
            .file_path = null,
        };
    }

    pub fn deinit(self: *EditorView) void {
        self.buffer.deinit();
        self.cursor.deinit();
        if (self.dirty_rows.len > 0) self.allocator.free(self.dirty_rows);
        if (self.file_path) |p| self.allocator.free(p);
    }

    pub fn updateViewport(self: *EditorView, width: u32, height: u32, font: *const FontFace) !void {
        const line_num_width: u32 = 4; // 4 digit columns for line numbers
        self.visible_cols = (width - self.left_pad) / font.cell_width - line_num_width - 1;
        self.visible_rows = height / font.cell_height;

        if (self.dirty_rows.len > 0) self.allocator.free(self.dirty_rows);
        self.dirty_rows = try self.allocator.alloc(bool, self.visible_rows);
        self.markAllDirty();
    }

    pub fn markAllDirty(self: *EditorView) void {
        @memset(self.dirty_rows, true);
    }

    pub fn markRowDirty(self: *EditorView, row: u32) void {
        if (row < self.dirty_rows.len) self.dirty_rows[row] = true;
    }

    /// Ensure cursor is visible, adjusting scroll if needed.
    pub fn ensureCursorVisible(self: *EditorView) void {
        const cursor_lc = self.buffer.offsetToLineCol(self.cursor.primary().head);
        if (cursor_lc.line < self.scroll_line) {
            self.scroll_line = cursor_lc.line;
            self.markAllDirty();
        } else if (cursor_lc.line >= self.scroll_line + self.visible_rows) {
            self.scroll_line = cursor_lc.line - self.visible_rows + 1;
            self.markAllDirty();
        }
    }

    /// Insert text at cursor, updating state.
    pub fn insertAtCursor(self: *EditorView, text: []const u8) !void {
        const pos = self.cursor.primary().head;
        try self.buffer.insert(pos, text);
        const new_pos = pos + @as(u32, @intCast(text.len));
        self.cursor.moveTo(new_pos);
        self.modified = true;

        // Mark affected rows dirty
        const lc = self.buffer.offsetToLineCol(pos);
        const screen_row = if (lc.line >= self.scroll_line) lc.line - self.scroll_line else 0;
        // If text contains newlines, dirty everything below
        if (std.mem.indexOfScalar(u8, text, '\n') != null) {
            var r = screen_row;
            while (r < self.dirty_rows.len) : (r += 1) self.dirty_rows[r] = true;
        } else {
            self.markRowDirty(screen_row);
        }
        self.ensureCursorVisible();
    }

    /// Delete selection or single char before cursor (backspace).
    pub fn backspace(self: *EditorView) !void {
        const sel = self.cursor.primary();
        if (sel.hasSelection()) {
            const start = sel.start();
            const len = sel.end() - start;
            try self.buffer.delete(start, len);
            self.cursor.moveTo(start);
        } else if (sel.head > 0) {
            // Delete one char before cursor (UTF-8 aware)
            self.cursor.moveLeft(&self.buffer, false);
            const new_pos = self.cursor.primary().head;
            try self.buffer.delete(new_pos, sel.head - new_pos);
        } else {
            return;
        }
        self.modified = true;
        self.markAllDirty(); // Simplified: mark all dirty for now
        self.ensureCursorVisible();
    }

    /// Delete char after cursor.
    pub fn deleteForward(self: *EditorView) !void {
        const sel = self.cursor.primary();
        if (sel.hasSelection()) {
            const start = sel.start();
            const len = sel.end() - start;
            try self.buffer.delete(start, len);
            self.cursor.moveTo(start);
        } else if (sel.head < self.buffer.total_len) {
            const slice = self.buffer.contiguousSliceAt(sel.head);
            const byte_len = CursorState.utf8ByteLen(slice[0]);
            try self.buffer.delete(sel.head, byte_len);
        } else {
            return;
        }
        self.modified = true;
        self.markAllDirty();
        self.ensureCursorVisible();
    }

    /// Render the visible portion of the document.
    pub fn render(self: *EditorView, renderer: *Renderer, font: *FontFace) void {
        const line_num_cols: u32 = 4;
        const gutter_width = (line_num_cols + 1) * font.cell_width + self.left_pad;
        const cursor_lc = self.buffer.offsetToLineCol(self.cursor.primary().head);
        const sel = self.cursor.primary();
        const sel_start = sel.start();
        const sel_end = sel.end();

        var line: u32 = self.scroll_line;
        var byte_offset = self.buffer.lineToOffset(self.scroll_line);
        var screen_row: u32 = 0;

        while (screen_row < self.visible_rows) : (screen_row += 1) {
            // Always advance byte_offset to track position, even for clean rows
            const this_line_start = byte_offset;
            _ = this_line_start;
            if (!self.dirty_rows[screen_row]) {
                // Skip clean rows — scan for next newline to advance byte_offset
                while (byte_offset < self.buffer.total_len) {
                    const slice = self.buffer.contiguousSliceAt(byte_offset);
                    if (slice.len == 0) break;
                    if (std.mem.indexOfScalar(u8, slice, '\n')) |nl_pos| {
                        byte_offset += @intCast(nl_pos + 1);
                        break;
                    }
                    byte_offset += @intCast(slice.len);
                }
                line += 1;
                continue;
            }
            self.dirty_rows[screen_row] = false;

            const row_y = screen_row * font.cell_height;

            // Clear row background
            renderer.fillRect(0, row_y, renderer.width, font.cell_height, bg_color);

            if (line >= self.buffer.lineCount()) {
                line += 1;
                continue;
            }

            // Draw line number
            const is_current_line = (line == cursor_lc.line);
            const ln_color = if (is_current_line) line_num_active_color else line_num_color;
            var ln_buf: [8]u8 = undefined;
            const ln_str = std.fmt.bufPrint(&ln_buf, "{d: >4}", .{line + 1}) catch "????";
            for (ln_str, 0..) |ch, ci| {
                renderer.drawCell(font, ch, @intCast(ci), screen_row, ln_color, bg_color, self.left_pad);
            }

            // Draw text content
            const next_line_offset = if (line + 1 < self.buffer.lineCount())
                self.buffer.lineToOffset(line + 1)
            else
                self.buffer.total_len;

            var col: u32 = 0;
            var off = byte_offset;
            while (off < next_line_offset) {
                const slice = self.buffer.contiguousSliceAt(off);
                if (slice.len == 0) break;
                const ch = slice[0];
                if (ch == '\n') {
                    off += 1;
                    break;
                }

                // Determine if this byte is in selection
                const in_selection = sel.hasSelection() and off >= sel_start and off < sel_end;
                const cell_bg = if (in_selection) selection_color else bg_color;

                if (ch == '\t') {
                    // Render tab as spaces to next tab stop
                    const tab_size: u32 = 4;
                    const spaces = tab_size - (col % tab_size);
                    var s: u32 = 0;
                    while (s < spaces) : (s += 1) {
                        renderer.drawCell(font, ' ', line_num_cols + 1 + col + s, screen_row, fg_color, cell_bg, self.left_pad);
                    }
                    col += spaces;
                    off += 1;
                } else {
                    const byte_len = CursorState.utf8ByteLen(ch);
                    // Decode UTF-8 codepoint
                    const cp = decodeUtf8(slice[0..@min(byte_len, @as(u32, @intCast(slice.len)))]);
                    renderer.drawCell(font, cp, line_num_cols + 1 + col, screen_row, fg_color, cell_bg, self.left_pad);
                    col += 1;
                    off += byte_len;
                }

                if (col >= self.visible_cols) break;
            }

            // Draw cursor (block cursor)
            if (is_current_line and cursor_lc.col <= self.visible_cols) {
                const cx = self.left_pad + (line_num_cols + 1 + cursor_lc.col) * font.cell_width;
                const cy = row_y;
                renderer.fillRect(cx, cy, 2, font.cell_height, cursor_color); // Thin line cursor
            }

            byte_offset = off;
            line += 1;
        }
    }

    fn decodeUtf8(bytes: []const u8) u32 {
        if (bytes.len == 0) return 0xFFFD;
        const b0 = bytes[0];
        if (b0 < 0x80) return b0;
        if (b0 < 0xE0 and bytes.len >= 2)
            return (@as(u32, b0 & 0x1F) << 6) | (bytes[1] & 0x3F);
        if (b0 < 0xF0 and bytes.len >= 3)
            return (@as(u32, b0 & 0x0F) << 12) | (@as(u32, bytes[1] & 0x3F) << 6) | (bytes[2] & 0x3F);
        if (bytes.len >= 4)
            return (@as(u32, b0 & 0x07) << 18) | (@as(u32, bytes[1] & 0x3F) << 12) | (@as(u32, bytes[2] & 0x3F) << 6) | (bytes[3] & 0x3F);
        return 0xFFFD;
    }
};

pub fn mapKey(keysym: u32, mods: Modifier) ?Action {

    // Parse CLI args
    const args = try std.process.argsAlloc(allocator);

    // Load file or start empty
    var content: []const u8 = "";
    var file_path: ?[]const u8 = null;
    if (args.len > 1) {
        file_path = args[1];

    }

        const n = std.posix.epoll_wait(epoll_fd, &events, -1);

        for (events[0..n]) |ev| {

fn handleKeyPress(editor: *EditorView, ke: Window.KeyEvent) bool {
    const action = keymap.mapKey(ke.keysym, ke.modifiers) orelse return false;

        .move_home => editor.cursor.moveHome(&editor.buffer, false),
        .move_end => editor.cursor.moveEnd(&editor.buffer, false),
        .select_left => editor.cursor.moveLeft(&editor.buffer, true),
        .select_right => editor.cursor.moveRight(&editor.buffer, true),
        .select_up => moveVertical(editor, -1, true),
        .select_down => moveVertical(editor, 1, true),
        .select_home => editor.cursor.moveHome(&editor.buffer, true),
        .select_end => editor.cursor.moveEnd(&editor.buffer, true),
        .select_all => {
            editor.cursor.cursors.items[0] = .{ .anchor = 0, .head = editor.buffer.total_len };
        },
        .backspace => editor.backspace() catch {},
        .delete => editor.deleteForward() catch {},
        .enter => editor.insertAtCursor("\n") catch {},
        .tab => editor.insertAtCursor("    ") catch {}, // insert spaces

        .copy => {}, // TODO: clipboard

    editor.ensureCursorVisible();

    if (lc.line >= editor.scroll_line) {
        editor.markRowDirty(lc.line - editor.scroll_line);
    }
    return true;
}

fn moveVertical(editor: *EditorView, delta: i32, extend: bool) void {
    const lc = editor.buffer.offsetToLineCol(editor.cursor.primary().head);
    const new_line = if (delta < 0)
        if (lc.line > 0) lc.line - 1 else 0
    else
        @min(lc.line + 1, editor.buffer.lineCount() - 1);

    if (new_line == lc.line) return;

    // Use sticky column (desired_col) for vertical movement
    const target_col = editor.cursor.desired_col orelse lc.col;
    if (editor.cursor.desired_col == null) {
        editor.cursor.desired_col = lc.col;
    }

    // Move to target column on new line (clamped to line length)
    const line_start = editor.buffer.lineToOffset(new_line);
    const next_line_start = editor.buffer.lineToOffset(new_line + 1);
    const line_len = if (next_line_start > line_start + 1)
        next_line_start - line_start - 1 // Exclude newline
    else if (next_line_start > line_start)
        next_line_start - line_start
    else
        0;

    const clamped_col = @min(target_col, line_len);
    const new_pos = line_start + clamped_col;

    if (extend) {
        editor.cursor.selectTo(new_pos);
    } else {
        editor.cursor.cursors.items[0] = .{ .anchor = new_pos, .head = new_pos };
        // Don't reset desired_col — preserve it for continued vertical movement
    }

    // Mark old and new rows dirty
    if (lc.line >= editor.scroll_line) editor.markRowDirty(lc.line - editor.scroll_line);
    if (new_line >= editor.scroll_line) editor.markRowDirty(new_line - editor.scroll_line);
}

fn handleScroll(editor: *EditorView, scroll: Window.ScrollEvent) void {
    const delta: i32 = scroll.delta * 3;
    if (delta < 0) {
        const up = @as(u32, @intCast(-delta));
        editor.scroll_line = if (editor.scroll_line > up) editor.scroll_line - up else 0;
    } else {
        editor.scroll_line = @min(
            editor.scroll_line + @as(u32, @intCast(delta)),
            editor.buffer.lineCount() -| 1,
        );
    }
    editor.markAllDirty();
}

fn saveFile(editor: *EditorView) void {
    const path = editor.file_path orelse return;
    // Atomic save: write to temp, rename
    const content = editor.buffer.collectContent(editor.allocator) catch return;
    defer editor.allocator.free(content);

    const dir = std.fs.cwd();
    const tmp_path = std.fmt.allocPrint(editor.allocator, "{s}.zz-tmp", .{path}) catch return;
    defer editor.allocator.free(tmp_path);

    const file = dir.createFile(tmp_path, .{}) catch return;
    file.writeAll(content) catch { file.close(); return; };
    file.close();

    dir.rename(tmp_path, path) catch return;
    editor.modified = false;
}

fn renderFrame(editor: *EditorView, win: *Window, font: *FontFace, cursor_visible: bool) void {
    _ = cursor_visible;
    var renderer = Renderer{
        .buffer = win.getBuffer(),
        .stride = win.stride,
        .width = win.width,
        .height = win.height,
    };
    editor.render(&renderer, font);
    win.present();
}

fn createTimerFd(interval_ns: u64) !std.posix.fd_t {
    const fd = try std.posix.timerfd_create(.MONOTONIC, .{ .NONBLOCK = true, .CLOEXEC = true });
    const ts = std.os.linux.timespec{
        .sec = @intCast(interval_ns / 1_000_000_000),
        .nsec = @intCast(interval_ns % 1_000_000_000),
    };
    const spec = std.os.linux.itimerspec{ .it_interval = ts, .it_value = ts };
    try std.posix.timerfd_settime(fd, .{}, &spec, null);
    return fd;
}

/// Caller must free the returned slice.
pub fn getSelectedText(self: *EditorView) ?[]u8 {
    const sel = self.cursor.primary();
    if (!sel.hasSelection()) return null;
    const start = sel.start();
    const len = sel.end() - start;

    var result = self.allocator.alloc(u8, len) catch return null;
    var offset: u32 = 0;
    var written: usize = 0;
    for (self.buffer.pieces.items) |piece| {
        const piece_end = offset + piece.len;
        defer offset = piece_end;

        if (piece_end <= start or offset >= sel.end()) continue;

        const content = self.buffer.pieceContent(piece);
        const local_start = if (offset < start) start - offset else 0;
        const local_end = if (piece_end > sel.end()) sel.end() - offset else piece.len;
        const slice = content[local_start..local_end];
        @memcpy(result[written..][0..slice.len], slice);
        written += slice.len;
    }
    return result;
}

pub fn deleteSelection(self: *EditorView) !void {
    const sel = self.cursor.primary();
    if (!sel.hasSelection()) return;
    const start = sel.start();
    const len = sel.end() - start;
    try self.buffer.delete(start, len);
    self.cursor.moveTo(start);
    self.modified = true;
    self.markAllDirty();
}

/// Convert pixel coordinates to buffer offset.
pub fn pixelToPosition(self: *EditorView, px: i32, py: i32, font: *const FontFace) u32 {
    const line_num_cols: u32 = 4;
    const gutter = (line_num_cols + 1) * font.cell_width + self.left_pad;

    const row: u32 = @intCast(@max(0, @divTrunc(py, @as(i32, @intCast(font.cell_height)))));
    const col: u32 = if (px > @as(i32, @intCast(gutter)))
        @intCast(@divTrunc(px - @as(i32, @intCast(gutter)), @as(i32, @intCast(font.cell_width))))
    else
        0;

    const target_line = self.scroll_line + row;
    if (target_line >= self.buffer.lineCount()) return self.buffer.total_len;

    const line_start = self.buffer.lineToOffset(target_line);
    // Walk col characters into the line
    var off = line_start;
    var c: u32 = 0;
    while (c < col and off < self.buffer.total_len) {
        const slice = self.buffer.contiguousSliceAt(off);
        if (slice.len == 0 or slice[0] == '\n') break;
        off += CursorState.utf8ByteLen(slice[0]);
        c += 1;
    }
    return off;
}
```
