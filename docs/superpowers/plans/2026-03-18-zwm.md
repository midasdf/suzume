# zwm Window Manager Implementation Plan

> **For agentic workers:** REQUIRED: Use superpowers:subagent-driven-development (if subagents available) or superpowers:executing-plans to implement this plan. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Build an ultra-minimal tiling window manager in Zig using xcb directly, with master+stack layout, dwm-style tags, multi-monitor support, and comptime configuration.

**Architecture:** Single-threaded blocking xcb event loop. Intrusive doubly-linked lists for client/monitor management. Pure-function layout engine. All configuration resolved at comptime via `config.zig`. EWMH/ICCCM atoms managed manually via `xcb_intern_atom`.

**Tech Stack:** Zig 0.15, xcb, xcb-keysyms, xcb-randr, libc

**Spec:** `docs/superpowers/specs/2026-03-18-zwm-design.md`

```zig
// zwm/src/xcb.zig
const std = @import("std");

pub const c = @cImport({
    @cInclude("xcb/xcb.h");
    @cInclude("xcb/xcb_keysyms.h");
    @cInclude("xcb/randr.h");
});

pub const Window = c.xcb_window_t;
pub const Connection = c.xcb_connection_t;
pub const Screen = c.xcb_screen_t;
pub const Atom = c.xcb_atom_t;
pub const Keycode = c.xcb_keycode_t;
pub const Keysym = c.xcb_keysym_t;
pub const Timestamp = c.xcb_timestamp_t;

pub const GenericEvent = c.xcb_generic_event_t;
pub const GenericError = c.xcb_generic_error_t;

// Event type constants (response_type & 0x7f)
pub const MAP_REQUEST = c.XCB_MAP_REQUEST;
pub const UNMAP_NOTIFY = c.XCB_UNMAP_NOTIFY;
pub const DESTROY_NOTIFY = c.XCB_DESTROY_NOTIFY;
pub const KEY_PRESS = c.XCB_KEY_PRESS;
pub const ENTER_NOTIFY = c.XCB_ENTER_NOTIFY;
pub const CONFIGURE_REQUEST = c.XCB_CONFIGURE_REQUEST;
pub const PROPERTY_NOTIFY = c.XCB_PROPERTY_NOTIFY;
pub const BUTTON_PRESS = c.XCB_BUTTON_PRESS;
pub const BUTTON_RELEASE = c.XCB_BUTTON_RELEASE;
pub const MOTION_NOTIFY = c.XCB_MOTION_NOTIFY;

/// Connect to X server. Returns connection and screen, or error.
pub fn connect() !struct { conn: *Connection, screen: *Screen } {
    var screen_num: c_int = 0;
    const conn = c.xcb_connect(null, &screen_num) orelse return error.XcbConnectFailed;

    if (c.xcb_connection_has_error(conn) != 0) {
        c.xcb_disconnect(conn);
        return error.XcbConnectFailed;
    }

    const setup = c.xcb_get_setup(conn);
    var iter = c.xcb_setup_roots_iterator(setup);

    var i: c_int = 0;
    while (i < screen_num) : (i += 1) {
        c.xcb_screen_next(&iter);
    }

    return .{ .conn = conn, .screen = iter.data };
}

/// Disconnect from X server.
pub fn disconnect(conn: *Connection) void {
    c.xcb_disconnect(conn);
}

/// Flush pending requests.
pub fn flush(conn: *Connection) void {
    _ = c.xcb_flush(conn);
}

/// Wait for next event (blocking).
pub fn waitForEvent(conn: *Connection) ?*GenericEvent {
    return c.xcb_wait_for_event(conn);
}

/// Get event type from generic event.
pub fn eventType(event: *GenericEvent) u8 {
    return event.response_type & 0x7f;
}

/// Intern an atom by name.
pub fn internAtom(conn: *Connection, name: []const u8, only_if_exists: bool) !Atom {
    const cookie = c.xcb_intern_atom(
        conn,
        @intFromBool(only_if_exists),
        @intCast(name.len),
        name.ptr,
    );
    const reply = c.xcb_intern_atom_reply(conn, cookie, null) orelse return error.InternAtomFailed;
    defer std.c.free(reply);
    return reply.atom;
}

/// Generate a new XID.
pub fn generateId(conn: *Connection) Window {
    return c.xcb_generate_id(conn);
}
```

```zig
pub fn tile(
    client_count: u32,
    mon: Geometry,
    master_count: u8,
    master_factor: f32,
    gap: u16,
    border: u16,
    buf: []Geometry,
) []Geometry {
    if (client_count == 0) return buf[0..0];
    const n: u32 = @min(client_count, @as(u32, buf.len));

    // Single client: fullscreen, no gaps/borders
    if (n == 1) {
        buf[0] = mon;
        return buf[0..1];
    }

    const mc: u32 = @min(@as(u32, master_count), n);
    const sc: u32 = n - mc; // stack count
    const total_border: u16 = border * 2;

    // Master area width
    const mw: u16 = if (sc == 0)
        mon.w
    else
        @intFromFloat(@as(f32, @floatFromInt(mon.w)) * master_factor);

    // Master windows — each gets equal vertical slice of master area
    const mh_total: i32 = @as(i32, mon.h);
    for (0..mc) |i| {
        const idx: u32 = @intCast(i);
        const y_start: i32 = @as(i32, mon.y) + @divTrunc(@as(i32, @intCast(idx)) * mh_total, @as(i32, @intCast(mc)));
        const y_end: i32 = @as(i32, mon.y) + @divTrunc(@as(i32, @intCast(idx + 1)) * mh_total, @as(i32, @intCast(mc)));
        buf[idx] = .{
            .x = mon.x,
            .y = @intCast(y_start),
            .w = mw -| total_border,
            .h = @intCast(y_end - y_start -| @as(i32, total_border)),
        };
    }

    // Stack windows — each gets equal vertical slice of stack area
    if (sc > 0) {
        const sx: i16 = @intCast(@as(i32, mon.x) + @as(i32, mw));
        const sw: u16 = mon.w - mw;

        for (0..sc) |i| {
            const idx: u32 = @intCast(i);
            const y_start: i32 = @as(i32, mon.y) + @divTrunc(@as(i32, @intCast(idx)) * mh_total, @as(i32, @intCast(sc)));
            const y_end: i32 = @as(i32, mon.y) + @divTrunc(@as(i32, @intCast(idx + 1)) * mh_total, @as(i32, @intCast(sc)));
            buf[mc + idx] = .{
                .x = sx,
                .y = @intCast(y_start),
                .w = sw -| total_border,
                .h = @intCast(y_end - y_start -| @as(i32, total_border)),
            };
        }
    }

    return buf[0..n];
}
```

```zig
// zwm/src/main.zig
const std = @import("std");
const xcb = @import("xcb.zig");
const config = @import("config.zig");
const layout = @import("layout.zig");

var running: bool = true;
var restart: bool = false;

fn sigHandler(sig: c_int) callconv(.c) void {
    _ = sig;
    running = false;
}

fn sighupHandler(sig: c_int) callconv(.c) void {
    _ = sig;
    running = false;
    restart = true;
}

pub fn main() !void {
    // Set up signal handlers
    const act = std.posix.Sigaction{
        .handler = .{ .handler = sigHandler },
        .mask = std.posix.empty_sigset,
        .flags = 0,
    };
    std.posix.sigaction(std.posix.SIG.TERM, &act, null) catch {};
    std.posix.sigaction(std.posix.SIG.INT, &act, null) catch {};

    // SIGHUP = restart (exec self)
    const hup_act = std.posix.Sigaction{
        .handler = .{ .handler = sighupHandler },
        .mask = std.posix.empty_sigset,
        .flags = 0,
    };
    std.posix.sigaction(std.posix.SIG.HUP, &hup_act, null) catch {};

    // Ignore SIGCHLD to auto-reap children
    const ign_act = std.posix.Sigaction{
        .handler = .{ .handler = std.posix.SIG.IGN },
        .mask = std.posix.empty_sigset,
        .flags = 0,
    };
    std.posix.sigaction(std.posix.SIG.CHLD, &ign_act, null) catch {};

    const stderr = std.io.getStdErr().writer();

    // Connect to X
    const x = xcb.connect() catch {
        try stderr.print("zwm: cannot open display\n", .{});
        return error.NoDisplay;
    };
    defer xcb.disconnect(x.conn);

    // Attempt to take over as WM (SubstructureRedirect on root)
    const values = [_]u32{
        xcb.c.XCB_EVENT_MASK_SUBSTRUCTURE_REDIRECT |
            xcb.c.XCB_EVENT_MASK_SUBSTRUCTURE_NOTIFY |
            xcb.c.XCB_EVENT_MASK_STRUCTURE_NOTIFY |
            xcb.c.XCB_EVENT_MASK_ENTER_WINDOW |
            xcb.c.XCB_EVENT_MASK_PROPERTY_CHANGE |
            xcb.c.XCB_EVENT_MASK_BUTTON_PRESS,
    };
    const cookie = xcb.c.xcb_change_window_attributes_checked(
        x.conn,
        x.screen.root,
```

```zig
pub const ClientList = struct {
    first: ?*Client = null,
    last: ?*Client = null,

    pub fn append(self: *ClientList, client: *Client) void {
        client.prev = self.last;
        client.next = null;
        if (self.last) |last| {
            last.next = client;
        } else {
            self.first = client;
        }
        self.last = client;
    }

    pub fn remove(self: *ClientList, client: *Client) void {
        if (client.prev) |prev| {
            prev.next = client.next;
        } else {
            self.first = client.next;
        }
        if (client.next) |next| {
            next.prev = client.prev;
        } else {
            self.last = client.prev;
        }
        client.prev = null;
        client.next = null;
    }

    /// Count visible clients matching tag mask.
    pub fn countVisible(self: *const ClientList, tag_mask: u9) u32 {
        var count: u32 = 0;
        var c = self.first;
        while (c) |client| : (c = client.next) {
            if (!client.is_floating and (client.tags & tag_mask) != 0) count += 1;
        }
        return count;
    }

    pub fn deinit(self: *Wm) void {
        // Free all clients
        var mon = self.monitors;
        while (mon) |m| {
            var c = m.clients.first;
            while (c) |client| {
                const next = client.next;
                self.alloc.destroy(client);
                c = next;
            }
            const next_mon = m.next;
            self.alloc.destroy(m);
            mon = next_mon;
        }

            .tags = mon.selected_tags,
            .monitor = mon,

        self.arrange(mon);
        xcb.flush(self.conn);

        const client = self.findClient(window) orelse return;
        const mon = client.monitor;

        if (self.focused == client) {
            self.focused = client.prev orelse client.next;
        }

        mon.clients.remove(client);
        self.alloc.destroy(client);
        self.arrange(mon);
        xcb.flush(self.conn);
    }

        var mon = self.monitors;

            var c = m.clients.first;
            while (c) |client| : (c = client.next) {
                if (client.window == window) return client;
            }
        }
        return null;
    }

    /// Re-tile all visible clients on a monitor.
    pub fn arrange(self: *Wm, mon: *Monitor) void {
        const visible_count = mon.clients.countVisible(mon.selected_tags);
        if (visible_count == 0) return;

        var buf: [64]layout.Geometry = undefined; // max 64 tiled windows

            config.gap_px,
            config.border_px,

            if (client.is_floating or (client.tags & mon.selected_tags) == 0) continue;
            if (idx >= geoms.len) break;

            const g = geoms[idx];

            client.y = g.y;

            client.h = g.h;

                    xcb.c.XCB_CONFIG_WINDOW_WIDTH |
                    xcb.c.XCB_CONFIG_WINDOW_HEIGHT |
                    xcb.c.XCB_CONFIG_WINDOW_BORDER_WIDTH,
```

```zig
    fn spawn(_: *Wm, argv: []const []const u8) void {
        if (argv.len == 0) return;
        const pid = std.posix.fork() catch return;
        if (pid == 0) {
            // Child: build null-terminated argv for execvp
            // Config strings are comptime literals, so .ptr is already null-terminated
            var c_argv: [16:null]?[*:0]const u8 = .{null} ** 16;
            for (argv, 0..) |arg, i| {
                if (i >= 15) break;
                c_argv[i] = @ptrCast(arg.ptr);
            }
            const envp: [*:null]const ?[*:0]const u8 = @ptrCast(std.c.environ);
            _ = std.c.execvpe(c_argv[0].?, @ptrCast(&c_argv), envp);
            std.posix.exit(1);
        }
    }

    fn focusNext(self: *Wm) void {
        const client = self.focused orelse return;
        const mon = client.monitor;
        // Find next visible client, wrapping around
        var c = client.next orelse mon.clients.first;
        while (c) |candidate| {
            if (candidate == client) break; // wrapped all the way around
            if ((candidate.tags & mon.selected_tags) != 0) {
                self.focused = candidate;
                self.updateFocus();
                return;
            }
            c = candidate.next orelse mon.clients.first;
        }

        const client = self.focused orelse return;
        const mon = client.monitor;
        var c = client.prev orelse mon.clients.last;
        while (c) |candidate| {
            if (candidate == client) break;
            if ((candidate.tags & mon.selected_tags) != 0) {
                self.focused = candidate;
                self.updateFocus();
                return;
            }
            c = candidate.prev orelse mon.clients.last;

    fn zoomClient(self: *Wm) void {
        const client = self.focused orelse return;
        const mon = client.monitor;
        if (client == mon.clients.first) return; // already master
        mon.clients.remove(client);
        // Prepend (make master)
        client.next = mon.clients.first;
        client.prev = null;
        if (mon.clients.first) |first| first.prev = client;
        mon.clients.first = client;
        if (mon.clients.last == null) mon.clients.last = client;
        self.arrange(mon);
        xcb.flush(self.conn);

    fn killClient(self: *Wm) void {
        const client = self.focused orelse return;
        // Try ICCCM WM_DELETE_WINDOW first
        if (self.supportsProtocol(client.window, self.wm_delete_window)) {
            var event: xcb.c.xcb_client_message_event_t = std.mem.zeroes(xcb.c.xcb_client_message_event_t);
            event.response_type = xcb.c.XCB_CLIENT_MESSAGE;
            event.window = client.window;
            event.type = self.wm_protocols;
            event.format = 32;
            event.data.data32[0] = self.wm_delete_window;
            event.data.data32[1] = xcb.c.XCB_CURRENT_TIME;
            xcb.c.xcb_send_event(self.conn, 0, client.window, xcb.c.XCB_EVENT_MASK_NO_EVENT, @ptrCast(&event));
        } else {
            // Forceful kill as fallback
            xcb.c.xcb_kill_client(self.conn, client.window);
        }
        xcb.flush(self.conn);
    }

    /// Check if a window supports a given WM_PROTOCOLS atom.
    fn supportsProtocol(self: *Wm, window: xcb.Window, proto: xcb.Atom) bool {
        const cookie = xcb.c.xcb_get_property(
            self.conn, 0, window, self.wm_protocols,
            xcb.c.XCB_ATOM_ATOM, 0, 32,
        );
        const reply = xcb.c.xcb_get_property_reply(self.conn, cookie, null) orelse return false;
        defer std.c.free(reply);
        const len = xcb.c.xcb_get_property_value_length(reply);
        const atoms: [*]const xcb.Atom = @ptrCast(@alignCast(xcb.c.xcb_get_property_value(reply)));
        const count: usize = @intCast(@divTrunc(len, @sizeOf(xcb.Atom)));
        for (0..count) |i| {
            if (atoms[i] == proto) return true;
        }
        return false;
    }

    fn toggleFloating(self: *Wm) void {
        const client = self.focused orelse return;
        client.is_floating = !client.is_floating;
        self.arrange(client.monitor);
        xcb.flush(self.conn);
    }

    fn toggleFullscreen(self: *Wm) void {
        const client = self.focused orelse return;
        client.is_fullscreen = !client.is_fullscreen;
        if (client.is_fullscreen) {
            const mon = client.monitor;
            const values = [_]u32{
                @intCast(@as(i32, mon.x)),
                @intCast(@as(i32, mon.y)),
                @intCast(@as(u32, mon.w)),
                @intCast(@as(u32, mon.h)),
                0, // no border
            };
            xcb.c.xcb_configure_window(
                self.conn,
                client.window,
                xcb.c.XCB_CONFIG_WINDOW_X | xcb.c.XCB_CONFIG_WINDOW_Y |
                    xcb.c.XCB_CONFIG_WINDOW_WIDTH | xcb.c.XCB_CONFIG_WINDOW_HEIGHT |
                    xcb.c.XCB_CONFIG_WINDOW_BORDER_WIDTH,
                &values,
            );
            // Raise above others
            const stack_values = [_]u32{xcb.c.XCB_STACK_MODE_ABOVE};
            xcb.c.xcb_configure_window(self.conn, client.window, xcb.c.XCB_CONFIG_WINDOW_STACK_MODE, &stack_values);
        } else {
            self.arrange(client.monitor);
        }
        xcb.flush(self.conn);
    }

    fn viewTag(self: *Wm, tag_index: u4) void {
        const mon = self.sel_monitor orelse return;
        const mask: u9 = @as(u9, 1) << @intCast(tag_index);
        if (mon.selected_tags == mask) return;
        mon.selected_tags = mask;

        // Hide windows not on this tag, show ones that are
        var c = mon.clients.first;
        while (c) |client| : (c = client.next) {
            if ((client.tags & mask) != 0) {
                xcb.c.xcb_map_window(self.conn, client.window);
            } else {
                xcb.c.xcb_unmap_window(self.conn, client.window);
            }
        }
        self.arrange(mon);
        self.focused = self.firstVisibleClient(mon);
        self.updateFocus();
    }

    fn tagClient(self: *Wm, tag_index: u4) void {
        const client = self.focused orelse return;
        const mask: u9 = @as(u9, 1) << @intCast(tag_index);
        client.tags = mask;
        const mon = client.monitor;
        if ((mask & mon.selected_tags) == 0) {
            // Client moved to invisible tag — hide it
```

```zig
const std = @import("std");
const xcb = @import("xcb.zig");
const config = @import("config.zig");

pub const MonitorInfo = struct {
    x: i16,
    y: i16,
    w: u16,
    h: u16,
};

/// Query XRandR for active monitors (CRTCs).
/// Returns a slice of MonitorInfo in the provided buffer.
pub fn detectMonitors(conn: *xcb.Connection, screen: *xcb.Screen, buf: []MonitorInfo) []MonitorInfo {
    const cookie = xcb.c.xcb_randr_get_screen_resources(conn, screen.root);
    const reply = xcb.c.xcb_randr_get_screen_resources_reply(conn, cookie, null) orelse {
        // Fallback: single monitor = full screen
        if (buf.len > 0) {
            buf[0] = .{ .x = 0, .y = 0, .w = screen.width_in_pixels, .h = screen.height_in_pixels };
            return buf[0..1];
        }
        return buf[0..0];
    };
    defer std.c.free(reply);

    const crtcs = xcb.c.xcb_randr_get_screen_resources_crtcs(reply);
    const num_crtcs = xcb.c.xcb_randr_get_screen_resources_crtcs_length(reply);

    var count: usize = 0;
    for (0..@intCast(num_crtcs)) |i| {
        if (count >= buf.len) break;
        const crtc_cookie = xcb.c.xcb_randr_get_crtc_info(conn, crtcs[i], 0);
        const crtc_reply = xcb.c.xcb_randr_get_crtc_info_reply(conn, crtc_cookie, null) orelse continue;
        defer std.c.free(crtc_reply);

        // Skip disabled CRTCs (width/height = 0)
        if (crtc_reply.width == 0 or crtc_reply.height == 0) continue;

        buf[count] = .{
            .x = crtc_reply.x,
            .y = crtc_reply.y,
            .w = crtc_reply.width,
            .h = crtc_reply.height,
        };
        count += 1;
    }

    // Fallback if no active CRTCs found
    if (count == 0 and buf.len > 0) {
        buf[0] = .{ .x = 0, .y = 0, .w = screen.width_in_pixels, .h = screen.height_in_pixels };
        return buf[0..1];
    }

    return buf[0..count];
}

        conn,
        screen.root,
        xcb.c.XCB_RANDR_NOTIFY_MASK_SCREEN_CHANGE,
    );
    xcb.flush(conn);
```

```zig
        const client = self.findClient(event.child) orelse return;

        // Focus on click
        self.focused = client;

        // Mod + Button1 = move, Mod + Button3 = resize
        const mod = event.state & 0xcf;
        if (mod != config.mod_key) return;

        if (event.detail == 1) { // Button1 = move

        self.grab_client_h = client.h;

            self.conn,
            0, // owner_events
            self.screen.root,

    pub fn handleMotionNotify(self: *Wm, event: *xcb.c.xcb_motion_notify_event_t) void {
        if (self.grab_type == .none) return;
        const client = if (self.grab_window) |w| self.findClient(w) else null;
        const c = client orelse return;

        const dx = event.root_x - self.grab_start_x;
        const dy = event.root_y - self.grab_start_y;

                c.y = self.grab_client_y + dy;

            },
            .none => return,
        }

        const values = [_]u32{
            @intCast(@as(i32, c.x)),
            @intCast(@as(i32, c.y)),
            @intCast(@as(u32, c.w)),
            @intCast(@as(u32, c.h)),
        };

            self.conn,
            c.window,
            xcb.c.XCB_CONFIG_WINDOW_X | xcb.c.XCB_CONFIG_WINDOW_Y |
                xcb.c.XCB_CONFIG_WINDOW_WIDTH | xcb.c.XCB_CONFIG_WINDOW_HEIGHT,
```

```zig
                const data: [*]const u32 = @ptrCast(@alignCast(xcb.c.xcb_get_property_value(hr)));
                const flags = data[0];
                // PMinSize (1<<4) and PMaxSize (1<<5) both set, and min==max
                if (flags & (1 << 4) != 0 and flags & (1 << 5) != 0) {
                    const min_w = data[5];
                    const min_h = data[6];
                    const max_w = data[7];
                    const max_h = data[8];
                    if (min_w == max_w and min_h == max_h) {
                        client.is_floating = true;
                        client.w = @intCast(min_w);
                        client.h = @intCast(min_h);
                    }
                }
            }
        }

        // Set border color
        const border_values = [_]u32{config.border_color};
        xcb.c.xcb_change_window_attributes(self.conn, window, xcb.c.XCB_CW_BORDER_PIXEL, &border_values);

        mon.clients.append(client);
        xcb.c.xcb_map_window(self.conn, window);

        self.focused = client;
        self.arrange(mon);
        self.updateFocus();
    }
```
