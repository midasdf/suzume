# ziawm Implementation Plan

> **For agentic workers:** REQUIRED: Use superpowers:subagent-driven-development (if subagents available) or superpowers:executing-plans to implement this plan. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Build an i3-compatible tiling window manager in Zig that is lighter and faster than i3.

**Architecture:** Externally i3-compatible (IPC, config syntax, i3bar protocol), internally optimized Zig. Container tree model with per-child percent sizing. Single-threaded epoll event loop. xcb direct (no Xlib). All core logic (tree, layout, config, command, criteria, IPC protocol) is xcb-independent and unit-testable.

**Tech Stack:** Zig (0.15.x), xcb, xcb-keysyms, xcb-randr, xcb-xkb, xkbcommon, libxft, fontconfig. Linux only (epoll, signalfd).

**Spec:** `docs/superpowers/specs/2026-03-19-ziawm-design.md`

```zig
// ziawm/src/tree.zig
const std = @import("std");

pub const ContainerType = enum {
    root,
    output,
    workspace,
    split_con,
    window,
};

pub const Layout = enum {
    hsplit,
    vsplit,
    tabbed,
    stacked,
};

pub const FullscreenMode = enum {
    none,
    window,
    global,
};

pub const Rect = struct {
    x: i32 = 0,
    y: i32 = 0,
    w: u32 = 0,
    h: u32 = 0,
};

pub const WindowData = struct {
    id: u32, // xcb window id
    class: []const u8 = "",
    instance: []const u8 = "",
    title: []const u8 = "",
    window_role: []const u8 = "",
    transient_for: ?u32 = null,
    urgency: bool = false,
};

pub const WorkspaceData = struct {
    name: []const u8,
    num: ?i32 = null,
    output_name: []const u8 = "",
};

pub const Container = struct {
    type: ContainerType,
    layout: Layout = .hsplit,

    // Tree structure
    parent: ?*Container = null,
    prev: ?*Container = null,
    next: ?*Container = null,
    children: ChildList = .{},

    // Geometry
    rect: Rect = .{},
    window_rect: Rect = .{},

    // Type-specific data
    window: ?WindowData = null,
    workspace: ?WorkspaceData = null,

    // Per-child sizing
    percent: f32 = 0.0, // 0.0 means "auto" (1.0 / num_siblings)

    // State
    is_floating: bool = false,
    is_fullscreen: FullscreenMode = .none,
    is_focused: bool = false,
    is_scratchpad: bool = false,
    dirty: bool = true,

    // Marks — simple bounded array, no heap alloc for typical use
    marks: [8]?[]const u8 = [_]?[]const u8{null} ** 8,
    mark_count: u8 = 0,

    pub fn create(allocator: *const std.mem.Allocator, container_type: ContainerType) !*Container {
        const self = try allocator.create(Container);
        self.* = .{ .type = container_type };
        return self;
    }

    pub fn destroy(self: *Container, allocator: *const std.mem.Allocator) void {
        // Unlink from parent first
        self.unlink();
        allocator.destroy(self);
    }

    /// Remove this container from its parent's child list
    pub fn unlink(self: *Container) void {
        if (self.parent) |p| {
            p.children.remove(self);
            self.parent = null;
        }
    }

    /// Append a child to this container
    pub fn appendChild(self: *Container, child: *Container) void {
        child.unlink();
        child.parent = self;
        self.children.append(child);
        self.dirty = true;
    }

    /// Insert child before the reference node
    pub fn insertBefore(self: *Container, child: *Container, ref: *Container) void {
        child.unlink();
        child.parent = self;
        self.children.insertBefore(child, ref);
        self.dirty = true;
    }

    /// Count visible (non-floating) children
    pub fn tilingChildCount(self: *const Container) usize {
        var count: usize = 0;
        var it = self.children.first;
        while (it) |c| : (it = c.next) {
            if (!c.is_floating) count += 1;
        }
        return count;
    }

    /// Find the focused child (first in child list = most recently focused)
    pub fn focusedChild(self: *const Container) ?*Container {
        return self.children.first;
    }

    /// Move child to head of children (mark as most recently focused)
    pub fn promoteChild(self: *Container, child: *Container) void {
        if (self.children.first == child) return;
        self.children.remove(child);
        self.children.prepend(child);
    }

    /// Add a mark
    pub fn addMark(self: *Container, mark: []const u8) bool {
        if (self.mark_count >= 8) return false;
        self.marks[self.mark_count] = mark;
        self.mark_count += 1;
        return true;
    }

    /// Remove a mark
    pub fn removeMark(self: *Container, mark: []const u8) bool {
        for (0..self.mark_count) |i| {
            if (self.marks[i]) |m| {
                if (std.mem.eql(u8, m, mark)) {
                    // Shift remaining marks
                    var j = i;
                    while (j + 1 < self.mark_count) : (j += 1) {
                        self.marks[j] = self.marks[j + 1];
                    }
                    self.marks[self.mark_count - 1] = null;
                    self.mark_count -= 1;
                    return true;
                }
            }
        }
        return false;
    }

    /// Check if container has a specific mark
    pub fn hasMark(self: *const Container, mark: []const u8) bool {
        for (0..self.mark_count) |i| {
            if (self.marks[i]) |m| {
                if (std.mem.eql(u8, m, mark)) return true;
            }
        }
        return false;
    }
};

/// Intrusive doubly-linked list of Container pointers
pub const ChildList = struct {
    first: ?*Container = null,
    last: ?*Container = null,

    pub fn append(self: *ChildList, node: *Container) void {
        node.prev = self.last;
        node.next = null;
        if (self.last) |last| {
            last.next = node;
        } else {
            self.first = node;
        }
        self.last = node;
    }

    pub fn prepend(self: *ChildList, node: *Container) void {
        node.next = self.first;
        node.prev = null;
        if (self.first) |first| {
            first.prev = node;
        } else {
            self.last = node;
        }
        self.first = node;
    }

    pub fn insertBefore(self: *ChildList, node: *Container, ref: *Container) void {
        node.next = ref;
        node.prev = ref.prev;
        if (ref.prev) |prev| {
            prev.next = node;
        } else {
            self.first = node;
        }
        ref.prev = node;
    }

    pub fn remove(self: *ChildList, node: *Container) void {
        if (node.prev) |prev| {
            prev.next = node.next;
        } else {
            self.first = node.next;
        }
        if (node.next) |next| {
            next.prev = node.prev;
        } else {
            self.last = node.prev;
        }
        node.prev = null;
        node.next = null;
    }

    pub fn len(self: *const ChildList) usize {
        var count: usize = 0;
        var it = self.first;
        while (it) |n| : (it = n.next) {
            count += 1;
        }
        return count;
    }
};
```

```zig
// ziawm/src/layout.zig
const std = @import("std");
const tree = @import("tree.zig");

/// Apply layout calculations to a container and its tiling children.
/// gap: gap between windows in pixels
/// border: border width in pixels
pub fn apply(con: *tree.Container, gap: u32, border: u32) void {
    // Collect tiling children
    var tiling: [64]*tree.Container = undefined;
    var count: usize = 0;
    var it = con.children.first;
    while (it) |c| : (it = c.next) {
        if (!c.is_floating and count < 64) {
            tiling[count] = c;
            count += 1;
        }
    }

    if (count == 0) return;

    const children = tiling[0..count];

    switch (con.layout) {
        .hsplit => applyHsplit(con.rect, children, gap, border),
        .vsplit => applyVsplit(con.rect, children, gap, border),
        .tabbed, .stacked => applyTabbed(con.rect, children, border),
    }

    // Recurse into split_con children
    for (children) |child| {
        if (child.type == .split_con) {
            apply(child, gap, border);
        }
    }
}

fn applyHsplit(area: tree.Rect, children: []*tree.Container, gap: u32, border: u32) void {
    const n: u32 = @intCast(children.len);
    if (n == 0) return;

    const total_gaps = if (n > 1) gap * (n - 1) else 0;
    const available_w = if (area.w > total_gaps) area.w - total_gaps else area.w;

    var x_offset: i16 = area.x;

    for (children, 0..) |child, i| {
        const child_w = getChildWidth(child, children, available_w, i);

        child.rect = .{
            .x = x_offset,
            .y = area.y,
            .w = child_w,
            .h = area.h,
        };
        child.window_rect = .{
            .x = x_offset + @as(i32, @intCast(border)),
            .y = area.y + @as(i32, @intCast(border)),
            .w = if (child_w > border * 2) child_w - border * 2 else child_w,
            .h = if (area.h > border * 2) area.h - border * 2 else area.h,
        };
        child.dirty = false;

        x_offset += @as(i32, @intCast(child_w));
        if (i + 1 < children.len) {
            x_offset += @as(i32, @intCast(gap));
        }
    }
}

fn applyVsplit(area: tree.Rect, children: []*tree.Container, gap: u32, border: u32) void {
    const n: u32 = @intCast(children.len);
    if (n == 0) return;

    const total_gaps = if (n > 1) gap * (n - 1) else 0;
    const available_h = if (area.h > total_gaps) area.h - total_gaps else area.h;

    var y_offset: i16 = area.y;

    for (children, 0..) |child, i| {
        const child_h = getChildHeight(child, children, available_h, i);

        child.rect = .{
            .x = area.x,
            .y = y_offset,
            .w = area.w,
            .h = child_h,
        };
        child.window_rect = .{
            .x = area.x + @as(i32, @intCast(border)),
            .y = y_offset + @as(i32, @intCast(border)),
            .w = if (area.w > border * 2) area.w - border * 2 else area.w,
            .h = if (child_h > border * 2) child_h - border * 2 else child_h,
        };
        child.dirty = false;

        y_offset += @as(i32, @intCast(child_h));
        if (i + 1 < children.len) {
            y_offset += @as(i32, @intCast(gap));
        }
    }
}

fn applyTabbed(area: tree.Rect, children: []*tree.Container, border: u32) void {
    // All children get the full area (only one is visible at a time)
    // Tab bar height reserved at top (16px per tab row)
    const tab_h: u32 = 16;
    for (children) |child| {
        child.rect = .{
            .x = area.x,
            .y = area.y + @as(i32, @intCast(tab_h)),
            .w = area.w,
            .h = if (area.h > tab_h) area.h - tab_h else area.h,
        };
        child.window_rect = .{
            .x = area.x + @as(i32, @intCast(border)),
            .y = area.y + @as(i32, @intCast(tab_h + border)),
            .w = if (area.w > border * 2) area.w - border * 2 else area.w,
            .h = if (area.h > tab_h + border * 2) area.h - tab_h - border * 2 else area.h,
        };
        child.dirty = false;
    }
}

/// Calculate width for a child in hsplit based on its percent
fn getChildWidth(child: *const tree.Container, siblings: []const *tree.Container, total: u32, index: usize) u32 {
    if (child.percent > 0.0) {
        return @intFromFloat(@as(f32, @floatFromInt(total)) * child.percent);
    }
    // Auto: equal distribution
    const n: u32 = @intCast(siblings.len);
    const base = total / n;
    // Give remainder to last child
    if (index == siblings.len - 1) {
        return total - base * (n - 1);
    }
    return base;
}

/// Calculate height for a child in vsplit based on its percent
fn getChildHeight(child: *const tree.Container, siblings: []const *tree.Container, total: u32, index: usize) u32 {
    return getChildWidth(child, siblings, total, index); // Same logic, different axis
}
```

```zig
// ziawm/src/criteria.zig
const std = @import("std");
const tree = @import("tree.zig");

pub const Criteria = struct {
    class: ?[]const u8 = null,
    instance: ?[]const u8 = null,
    title: ?[]const u8 = null,
    window_role: ?[]const u8 = null,
    con_mark: ?[]const u8 = null,
    floating: ?bool = null,
    workspace: ?[]const u8 = null,
};

/// Parse "[class=\"Firefox\" title=\"Git*\"]" into a Criteria struct.
/// Input must start with '[' and end with ']'.
/// Returns null if parse fails.
pub fn parse(input: []const u8) ?Criteria {
    if (input.len < 2 or input[0] != '[' or input[input.len - 1] != ']') return null;

    var result = Criteria{};
    var pos: usize = 1; // skip '['
    const end = input.len - 1; // before ']'

    while (pos < end) {
        // Skip whitespace
        while (pos < end and input[pos] == ' ') pos += 1;
        if (pos >= end) break;

        // Read key
        const key_start = pos;
        while (pos < end and input[pos] != '=') pos += 1;
        if (pos >= end) return null;
        const key = input[key_start..pos];
        pos += 1; // skip '='

        // Read value (quoted)
        if (pos >= end or input[pos] != '"') return null;
        pos += 1; // skip opening '"'
        const val_start = pos;
        while (pos < end and input[pos] != '"') pos += 1;
        if (pos >= end) return null;
        const val = input[val_start..pos];
        pos += 1; // skip closing '"'

        // Assign to field
        if (std.mem.eql(u8, key, "class")) {
            result.class = val;
        } else if (std.mem.eql(u8, key, "instance")) {
            result.instance = val;
        } else if (std.mem.eql(u8, key, "title")) {
            result.title = val;
        } else if (std.mem.eql(u8, key, "window_role")) {
            result.window_role = val;
        } else if (std.mem.eql(u8, key, "con_mark")) {
            result.con_mark = val;
        } else if (std.mem.eql(u8, key, "floating")) {
            result.floating = std.mem.eql(u8, val, "true") or std.mem.eql(u8, val, "yes");
        } else if (std.mem.eql(u8, key, "workspace")) {
            result.workspace = val;
        }
        // Unknown keys are silently ignored
    }

    return result;
}

/// Glob match: only '*' wildcard supported.
/// '*' matches zero or more of any character.
pub fn globMatch(pattern: []const u8, text: []const u8) bool {
    var pi: usize = 0;
    var ti: usize = 0;
    var star_pi: ?usize = null;
    var star_ti: usize = 0;

    while (ti < text.len) {
        if (pi < pattern.len and (pattern[pi] == text[ti] or pattern[pi] == '?')) {
            pi += 1;
            ti += 1;
        } else if (pi < pattern.len and pattern[pi] == '*') {
            star_pi = pi;
            star_ti = ti;
            pi += 1;
        } else if (star_pi) |sp| {
            pi = sp + 1;
            star_ti += 1;
            ti = star_ti;
        } else {
            return false;
        }
    }

    while (pi < pattern.len and pattern[pi] == '*') pi += 1;
    return pi == pattern.len;
}

/// Check if a container matches the given criteria
pub fn matches(crit: *const Criteria, con: *const tree.Container) bool {
    if (crit.class) |pattern| {
        if (con.window) |w| {
            if (!globMatch(pattern, w.class)) return false;
        } else return false;
    }
    if (crit.instance) |pattern| {
        if (con.window) |w| {
            if (!globMatch(pattern, w.instance)) return false;
        } else return false;
    }
    if (crit.title) |pattern| {
        if (con.window) |w| {
            if (!globMatch(pattern, w.title)) return false;
        } else return false;
    }
    if (crit.window_role) |pattern| {
        if (con.window) |w| {
            if (!globMatch(pattern, w.window_role)) return false;
        } else return false;
    }
    if (crit.con_mark) |mark| {
        if (!con.hasMark(mark)) return false;
    }
    if (crit.floating) |expected| {
        if (con.is_floating != expected) return false;
    }
    // workspace matching requires walking up to workspace parent — skip for now
    return true;
}
```

```zig
// ziawm/src/ipc.zig
const std = @import("std");

pub const MAGIC = "i3-ipc";
pub const HEADER_SIZE = MAGIC.len + 4 + 4; // 14 bytes

pub const MessageType = enum(u32) {
    run_command = 0,
    get_workspaces = 1,
    subscribe = 2,
    get_outputs = 3,
    get_tree = 4,
    get_marks = 5,
    get_bar_config = 6,
    get_version = 7,
    send_tick = 8,
    get_config = 9,
    get_binding_modes = 10,
};

/// Event types have bit 31 set
pub const EventType = enum(u32) {
    workspace = 0x80000000,
    output = 0x80000001,
    mode = 0x80000002,
    window = 0x80000003,
    barconfig_update = 0x80000004,
    binding = 0x80000005,
};

pub const MAX_PAYLOAD_SIZE = 1024 * 1024; // 1MB max payload

/// Encode a message into i3-ipc wire format.
/// Writes into caller-provided buffer, returns the filled slice.
pub fn encode(msg_type: MessageType, payload: []const u8, buf: []u8) []const u8 {
    const total = HEADER_SIZE + payload.len;
    std.debug.assert(buf.len >= total);

    @memcpy(buf[0..6], MAGIC);
    std.mem.writeInt(u32, buf[6..10], @intCast(payload.len), .little);
    std.mem.writeInt(u32, buf[10..14], @intFromEnum(msg_type), .little);
    if (payload.len > 0) {
        @memcpy(buf[14 .. 14 + payload.len], payload);
    }

    return buf[0..total];
}

/// Encode an event message (bit 31 set in type)
pub fn encodeEvent(event_type: EventType, payload: []const u8, buf: []u8) []const u8 {
    const total = HEADER_SIZE + payload.len;
    std.debug.assert(buf.len >= total);

    @memcpy(buf[0..6], MAGIC);
    std.mem.writeInt(u32, buf[6..10], @intCast(payload.len), .little);
    std.mem.writeInt(u32, buf[10..14], @intFromEnum(event_type), .little);
    if (payload.len > 0) {
        @memcpy(buf[14 .. 14 + payload.len], payload);
    }

    return buf[0..total];
}

/// Decoded message header
pub const Header = struct {
    msg_type: u32,
    payload_len: u32,
};

/// Decode header from buffer. Returns null if buffer too small or invalid magic.
pub fn decodeHeader(buf: []const u8) ?Header {
    if (buf.len < HEADER_SIZE) return null;
    if (!std.mem.eql(u8, buf[0..6], MAGIC)) return null;

    return .{
        .payload_len = std.mem.readInt(u32, buf[6..10], .little),
        .msg_type = std.mem.readInt(u32, buf[10..14], .little),
    };
}

/// Check if a message type value represents an event (bit 31 set)
pub fn isEvent(msg_type: u32) bool {
    return (msg_type & 0x80000000) != 0;
}
```

```zig
const std = @import("std");
const criteria_mod = @import("criteria.zig");

pub const Modifier = enum {
    mod1, // Alt
    mod4, // Super
    shift,
    control,
};

pub const Keybind = struct {
    modifiers: u8 = 0, // bitmask: bit0=mod, bit1=shift, bit2=control
    key: []const u8,
    command: []const u8,
    mode: []const u8 = "default",
};

pub const BarConfig = struct {
    status_command: []const u8 = "",
    position: []const u8 = "bottom",
    bg_color: []const u8 = "#222222",
    statusline_color: []const u8 = "#dddddd",
};

pub const WindowRule = struct {
    criteria: criteria_mod.Criteria,
    command: []const u8,
};

pub const AssignRule = struct {
    criteria: criteria_mod.Criteria,
    workspace: []const u8,
};

pub const Config = struct {
    allocator: *const std.mem.Allocator,

    // Variables
    variables: std.StringHashMap([]const u8),

    // Keybinds per mode
    keybinds: std.ArrayList(Keybind),

    // Appearance
    border_px: u32 = 1,
    gap_inner: u32 = 0,
    gap_outer: u32 = 0,
    focus_follows_mouse: bool = true,
    focus_wrapping: bool = true,

    focused_bg: []const u8 = "#285577",
    focused_text: []const u8 = "#ffffff",
    unfocused_border: []const u8 = "#333333",
    unfocused_bg: []const u8 = "#222222",
    unfocused_text: []const u8 = "#888888",

    window_rules: std.ArrayList(WindowRule),

    assign_rules: std.ArrayList(AssignRule),

    modes: std.StringHashMap(void), // just track which modes exist

    // Raw config text (for GET_CONFIG IPC)
    raw_text: []const u8 = "",

    pub fn parse(allocator: *const std.mem.Allocator, text: []const u8) !Config {
        var cfg = Config{
            .allocator = allocator,
            .variables = std.StringHashMap([]const u8).init(allocator.*),
            .keybinds = std.ArrayList(Keybind).init(allocator.*),
            .window_rules = std.ArrayList(WindowRule).init(allocator.*),
            .assign_rules = std.ArrayList(AssignRule).init(allocator.*),
            .exec_commands = std.ArrayList([]const u8).init(allocator.*),
            .exec_always_commands = std.ArrayList([]const u8).init(allocator.*),
            .workspace_outputs = std.StringHashMap([]const u8).init(allocator.*),
            .modes = std.StringHashMap(void).init(allocator.*),
            .raw_text = text,
        };

        var current_mode: []const u8 = "default";
        var lines = std.mem.splitScalar(u8, text, '\n');

        while (lines.next()) |raw_line| {
            const line = std.mem.trim(u8, raw_line, " \t\r");
            if (line.len == 0 or line[0] == '#') continue;

            // Expand variables
            const expanded = cfg.expandVars(line);

            if (std.mem.startsWith(u8, expanded, "set ")) {
                cfg.parseSet(expanded) catch continue;
            } else if (std.mem.startsWith(u8, expanded, "bindsym ")) {
                cfg.parseBindsym(expanded, current_mode) catch continue;
            } else if (std.mem.startsWith(u8, expanded, "default_border pixel ")) {
                cfg.border_px = std.fmt.parseInt(u32, expanded["default_border pixel ".len..], 10) catch 1;
            } else if (std.mem.startsWith(u8, expanded, "gaps inner ")) {
                cfg.gap_inner = std.fmt.parseInt(u32, expanded["gaps inner ".len..], 10) catch 0;
            } else if (std.mem.startsWith(u8, expanded, "gaps outer ")) {
                cfg.gap_outer = std.fmt.parseInt(u32, expanded["gaps outer ".len..], 10) catch 0;
            } else if (std.mem.startsWith(u8, expanded, "focus_follows_mouse ")) {
                const val = expanded["focus_follows_mouse ".len..];
                cfg.focus_follows_mouse = std.mem.eql(u8, val, "yes") or std.mem.eql(u8, val, "true");
            } else if (std.mem.startsWith(u8, expanded, "focus_wrapping ")) {
                const val = expanded["focus_wrapping ".len..];
                cfg.focus_wrapping = std.mem.eql(u8, val, "yes") or std.mem.eql(u8, val, "true");
            } else if (std.mem.startsWith(u8, expanded, "for_window ")) {
                cfg.parseForWindow(expanded) catch continue;
            } else if (std.mem.startsWith(u8, expanded, "assign ")) {
                cfg.parseAssign(expanded) catch continue;
            } else if (std.mem.startsWith(u8, expanded, "exec_always ")) {
                const cmd = stripNoStartupId(expanded["exec_always ".len..]);
                try cfg.exec_always_commands.append(cmd);
            } else if (std.mem.startsWith(u8, expanded, "exec ")) {
                const cmd = stripNoStartupId(expanded["exec ".len..]);
                try cfg.exec_commands.append(cmd);
            } else if (std.mem.startsWith(u8, expanded, "mode \"")) {
                // Parse mode name
                const name_start = "mode \"".len;
                const name_end = std.mem.indexOfScalar(u8, expanded[name_start..], '"') orelse continue;
                current_mode = expanded[name_start .. name_start + name_end];
                try cfg.modes.put(current_mode, {});
            } else if (std.mem.eql(u8, expanded, "}")) {
                current_mode = "default";
            } else if (std.mem.startsWith(u8, expanded, "bar {")) {
                // Bar block handled by scanning ahead — simplified for now
                // TODO: proper bar block parsing in next iteration
            } else if (std.mem.startsWith(u8, expanded, "status_command ")) {
                cfg.bar.status_command = expanded["status_command ".len..];
            } else if (std.mem.startsWith(u8, expanded, "position ")) {
                cfg.bar.position = expanded["position ".len..];
            } else if (std.mem.startsWith(u8, expanded, "workspace ")) {
                cfg.parseWorkspaceOutput(expanded) catch continue;
            } else if (std.mem.startsWith(u8, expanded, "client.focused ")) {
                cfg.parseClientColor(expanded["client.focused ".len..], &cfg.focused_border, &cfg.focused_bg, &cfg.focused_text);
            } else if (std.mem.startsWith(u8, expanded, "client.unfocused ")) {
                cfg.parseClientColor(expanded["client.unfocused ".len..], &cfg.unfocused_border, &cfg.unfocused_bg, &cfg.unfocused_text);
            } else if (std.mem.startsWith(u8, expanded, "background ")) {
                cfg.bar.bg_color = expanded["background ".len..];
            } else if (std.mem.startsWith(u8, expanded, "statusline ")) {
                cfg.bar.statusline_color = expanded["statusline ".len..];
            }
            // Unknown lines silently skipped
        }

        return cfg;
    }

    fn parseSet(self: *Config, line: []const u8) !void {
        // "set $var value"
        const rest = line["set ".len..];
        const space = std.mem.indexOfScalar(u8, rest, ' ') orelse return error.InvalidFormat;
        const name = rest[0..space];
        const value = rest[space + 1 ..];
        try self.variables.put(name, value);
    }

    fn parseBindsym(self: *Config, line: []const u8, mode: []const u8) !void {
        // "bindsym Mod4+Shift+Return exec st"
        const rest = line["bindsym ".len..];
        const space = std.mem.indexOfScalar(u8, rest, ' ') orelse return error.InvalidFormat;
        const key_combo = rest[0..space];
        const command = rest[space + 1 ..];

        var modifiers: u8 = 0;
        var key: []const u8 = key_combo;

        // Parse modifier+key combinations
        var parts = std.mem.splitScalar(u8, key_combo, '+');
        var last_part: []const u8 = "";
        while (parts.next()) |part| {
            if (std.mem.eql(u8, last_part, "")) {
                // skip
            } else if (isModifier(last_part)) {
                modifiers |= modifierBit(last_part);
            }
            last_part = part;
        }
        key = last_part; // Last part is the actual key

        try self.keybinds.append(.{
            .modifiers = modifiers,
            .key = key,
            .command = command,
            .mode = mode,
        });
    }

    fn parseForWindow(self: *Config, line: []const u8) !void {
        // "for_window [class=\"mpv\"] floating enable"
        const rest = line["for_window ".len..];
        const bracket_end = std.mem.indexOfScalar(u8, rest, ']') orelse return error.InvalidFormat;
        const criteria_str = rest[0 .. bracket_end + 1];
        const command = std.mem.trim(u8, rest[bracket_end + 1 ..], " ");
        const crit = criteria_mod.parse(criteria_str) orelse return error.InvalidCriteria;
        try self.window_rules.append(.{ .criteria = crit, .command = command });
    }

    fn parseAssign(self: *Config, line: []const u8) !void {
        // "assign [class=\"Firefox\"] workspace 2"
        const rest = line["assign ".len..];
        const bracket_end = std.mem.indexOfScalar(u8, rest, ']') orelse return error.InvalidFormat;
        const criteria_str = rest[0 .. bracket_end + 1];
        var ws_text = std.mem.trim(u8, rest[bracket_end + 1 ..], " ");
        if (std.mem.startsWith(u8, ws_text, "workspace ")) {
            ws_text = ws_text["workspace ".len..];
        }
        const crit = criteria_mod.parse(criteria_str) orelse return error.InvalidCriteria;
        try self.assign_rules.append(.{ .criteria = crit, .workspace = ws_text });
    }

    fn parseWorkspaceOutput(self: *Config, line: []const u8) !void {
        // "workspace 1 output HDMI-1"
        const rest = line["workspace ".len..];
        // Find "output" keyword
        if (std.mem.indexOf(u8, rest, " output ")) |idx| {
            const ws_name = rest[0..idx];
            const output_name = rest[idx + " output ".len ..];
            try self.workspace_outputs.put(ws_name, output_name);
        }
    }

    /// Expand $variables in a line. Uses a static buffer (max 1024 chars per line).
    fn expandVars(self: *const Config, line: []const u8) []const u8 {
        if (std.mem.indexOfScalar(u8, line, '$') == null) return line;

        const S = struct {
            var buf: [1024]u8 = undefined;
        };
        var out_len: usize = 0;
        var i: usize = 0;

        while (i < line.len) {
            if (line[i] == '$') {
                // Find end of variable name (alphanumeric + _)
                const var_start = i;
                i += 1;
                while (i < line.len and (std.ascii.isAlphanumeric(line[i]) or line[i] == '_')) : (i += 1) {}
                const var_name = line[var_start..i];
                if (self.variables.get(var_name)) |value| {
                    if (out_len + value.len <= S.buf.len) {
                        @memcpy(S.buf[out_len .. out_len + value.len], value);
                        out_len += value.len;
                    }
                } else {
                    // Unknown variable, keep as-is
                    if (out_len + var_name.len <= S.buf.len) {
                        @memcpy(S.buf[out_len .. out_len + var_name.len], var_name);
                        out_len += var_name.len;
                    }
                }
            } else {
                if (out_len < S.buf.len) {
                    S.buf[out_len] = line[i];
                    out_len += 1;
                }
                i += 1;
            }
        }

        return S.buf[0..out_len];
    }

    fn parseClientColor(self: *Config, rest: []const u8, border: *[]const u8, bg: *[]const u8, text: *[]const u8) void {
        _ = self;
        // "border bg text indicator child_border" — we take first 3
        var parts = std.mem.splitScalar(u8, rest, ' ');
        if (parts.next()) |b| border.* = b;
        if (parts.next()) |b| bg.* = b;
        if (parts.next()) |t| text.* = t;
    }

    fn stripNoStartupId(cmd: []const u8) []const u8 {
        if (std.mem.startsWith(u8, cmd, "--no-startup-id ")) {
            return cmd["--no-startup-id ".len..];
        }
        return cmd;
    }

    pub fn getVariable(self: *const Config, name: []const u8) ?[]const u8 {
        return self.variables.get(name);
    }

    pub fn deinit(self: *Config) void {
        self.variables.deinit();
        self.keybinds.deinit();
        self.window_rules.deinit();
        self.assign_rules.deinit();
        self.exec_commands.deinit();
        self.exec_always_commands.deinit();
        self.workspace_outputs.deinit();
        self.modes.deinit();
    }
};

fn isModifier(s: []const u8) bool {
    return std.mem.eql(u8, s, "Mod4") or std.mem.eql(u8, s, "Mod1") or
        std.mem.eql(u8, s, "Shift") or std.mem.eql(u8, s, "Control");
}

fn modifierBit(s: []const u8) u8 {
    if (std.mem.eql(u8, s, "Mod4")) return 1;  // Super
    if (std.mem.eql(u8, s, "Mod1")) return 8;  // Alt
    if (std.mem.eql(u8, s, "Shift")) return 2;
    if (std.mem.eql(u8, s, "Control")) return 4;
    return 0;
}
```

```zig
// ziawm/src/command.zig
const std = @import("std");
const tree = @import("tree.zig");
const criteria_mod = @import("criteria.zig");

pub const CommandType = enum {
    split,
    focus,
    move,
    layout_cmd,
    resize,
    workspace,
    move_workspace,
    kill,
    exec,
    floating,
    fullscreen,
    mark,
    unmark,
    scratchpad,
    mode,
    reload,
    restart,
    exit,
    focus_output,
    nop,
};

pub const Command = struct {
    type: CommandType,
    args: [4]?[]const u8 = [_]?[]const u8{null} ** 4,
    criteria: ?criteria_mod.Criteria = null,
};

/// Parse an i3 command string into a Command struct.
/// Supports: split h/v, focus left/right/up/down/parent/child/mode_toggle,
/// move left/right/up/down, layout stacking/tabbed/toggle split,
/// resize grow/shrink width/height N px, workspace N/name,
/// move container to workspace N, kill, exec cmd, floating enable/disable/toggle,
/// fullscreen toggle, mark name, unmark name, scratchpad show,
/// move scratchpad, mode name, reload, restart, exit
pub fn parse(input: []const u8) ?Command {
    const trimmed = std.mem.trim(u8, input, " \t");
    if (trimmed.len == 0) return null;

    // Check for criteria prefix: [class="X"] command
    var cmd_text = trimmed;
    var crit: ?criteria_mod.Criteria = null;
    if (trimmed[0] == '[') {
        if (std.mem.indexOfScalar(u8, trimmed, ']')) |end| {
            crit = criteria_mod.parse(trimmed[0 .. end + 1]);
            cmd_text = std.mem.trim(u8, trimmed[end + 1 ..], " ");
        }
    }

    var result = parseCommand(cmd_text) orelse return null;
    result.criteria = crit;
    return result;
}

fn parseCommand(text: []const u8) ?Command {
    if (std.mem.startsWith(u8, text, "split ")) {
        return .{ .type = .split, .args = makeArgs(text["split ".len..], null, null, null) };
    } else if (std.mem.startsWith(u8, text, "focus output ")) {
        return .{ .type = .focus_output, .args = makeArgs(text["focus output ".len..], null, null, null) };
    } else if (std.mem.startsWith(u8, text, "focus ")) {
        return .{ .type = .focus, .args = makeArgs(text["focus ".len..], null, null, null) };
    } else if (std.mem.startsWith(u8, text, "move container to workspace number ")) {
        return .{ .type = .move_workspace, .args = makeArgs(text["move container to workspace number ".len..], null, null, null) };
    } else if (std.mem.startsWith(u8, text, "move container to workspace ")) {
        return .{ .type = .move_workspace, .args = makeArgs(text["move container to workspace ".len..], null, null, null) };
    } else if (std.mem.startsWith(u8, text, "move scratchpad")) {
        return .{ .type = .scratchpad, .args = makeArgs("move", null, null, null) };

        return .{ .type = .exit };
    } else if (std.mem.eql(u8, text, "nop")) {
        return .{ .type = .nop };
    }

fn parseResize(text: []const u8) ?Command {
    // "grow width 10 px" or "shrink height 5 px"
    var parts = std.mem.splitScalar(u8, text, ' ');
    const direction = parts.next() orelse return null; // grow/shrink
    const dimension = parts.next() orelse return null; // width/height
    const amount = parts.next() orelse return null;    // number
    const unit = parts.next();                          // px (optional)
    return .{
        .type = .resize,
        .args = makeArgs(direction, dimension, amount, if (unit) |u| u else "px"),
    };
}

fn stripQuotes(s: []const u8) []const u8 {
    if (s.len >= 2 and s[0] == '"' and s[s.len - 1] == '"') {
        return s[1 .. s.len - 1];
    }
    return s;
}

fn makeArgs(a0: ?[]const u8, a1: ?[]const u8, a2: ?[]const u8, a3: ?[]const u8) [4]?[]const u8 {
    return .{ a0, a1, a2, a3 };
}
```

```zig
pub fn findByName(root: *tree.Container, name: []const u8) ?*tree.Container {
    var out_it = root.children.first;
    while (out_it) |output| : (out_it = output.next) {
        var ws_it = output.children.first;
        while (ws_it) |ws| : (ws_it = ws.next) {
            if (ws.workspace) |ws_data| {
                if (std.mem.eql(u8, ws_data.name, name)) return ws;
            }
        }
    }
    return null;
}

/// Find workspace by number in the tree
pub fn findByNum(root: *tree.Container, num: i32) ?*tree.Container {
    var out_it = root.children.first;
    while (out_it) |output| : (out_it = output.next) {
        var ws_it = output.children.first;
        while (ws_it) |ws| : (ws_it = ws.next) {
            if (ws.workspace) |ws_data| {
                if (ws_data.num) |n| {
                    if (n == num) return ws;
                }
            }
        }
    }
    return null;
}
```

```zig
// ziawm/src/render.zig
const std = @import("std");
const xcb = @import("xcb.zig");
const tree = @import("tree.zig");

/// Apply layout coordinates from the tree to actual X windows
pub fn applyTree(conn: *xcb.Connection, root: *tree.Container, border_focus: u32, border_unfocus: u32) void {
    var it = root.children.first;
    while (it) |output| : (it = output.next) {
        var ws_it = output.children.first;
        while (ws_it) |ws| : (ws_it = ws.next) {
            applyContainer(conn, ws, border_focus, border_unfocus);
        }
    }
}

fn applyContainer(conn: *xcb.Connection, con: *tree.Container, border_focus: u32, border_unfocus: u32) void {
    if (con.type == .window) {
        if (con.window) |w| {
            configureWindow(conn, w.id, con.window_rect, 0);
            setBorderColor(conn, w.id, if (con.is_focused) border_focus else border_unfocus);
            xcb.mapWindow(conn, w.id);
        }
        return;
    }

    // For tabbed/stacked layouts: only show the focused child, unmap others
    const is_tabbed = (con.layout == .tabbed or con.layout == .stacked);
    const focused = if (is_tabbed) con.focusedChild() else null;

    // Recurse into children
    var it = con.children.first;
    while (it) |child| : (it = child.next) {
        if (is_tabbed and child != focused) {
            // Unmap non-focused tabs
            if (child.type == .window) {
                if (child.window) |w| xcb.unmapWindow(conn, w.id);
            }
        } else {
            applyContainer(conn, child, border_focus, border_unfocus);
```

```zig
/// Create and bind the IPC listening socket
pub fn createServer(socket_path: []const u8) !posix.fd_t {
    // Ensure directory exists
    if (std.fs.path.dirname(socket_path)) |dir| {
        std.fs.makeDirAbsolute(dir) catch |e| switch (e) {
            error.PathAlreadyExists => {},
            else => return e,
        };
    }

    // Remove stale socket
    std.fs.deleteFileAbsolute(socket_path) catch {};

    const fd = try posix.socket(posix.AF.UNIX, posix.SOCK.STREAM | posix.SOCK.NONBLOCK | posix.SOCK.CLOEXEC, 0);

    var addr = std.net.Address.initUnix(socket_path) catch return error.InvalidPath;
    try posix.bind(fd, &addr.any, addr.getOsSockLen());
    try posix.listen(fd, 5);

    return fd;
}

pub fn readMessage(fd: posix.fd_t, buf: []u8) !?struct { msg_type: u32, payload: []const u8 } {

    if (n < HEADER_SIZE + header.payload_len) return null;

        .msg_type = header.msg_type,
        .payload = buf[HEADER_SIZE .. HEADER_SIZE + header.payload_len],
```

```zig
        } else {
            payload = args[i];
        }

    const path = socket_path orelse blk: {
```
